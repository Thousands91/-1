echo "=== $(date '+%H:%M:%S') 镜像测速+模型部署 ==="
MODEL="k2-fsa/sherpa-onnx/releases/download/asr-models/sherpa-onnx-paraformer-zh-small-2024-03-09.tar.bz2"
BEST=""; BESTSP=0
for m in "https://ghfast.top/https://github.com" "https://gh-proxy.com/https://github.com" "https://ghproxy.net/https://github.com" "https://github.com"; do
  sp=$(curl -sL -m 30 -r 0-2097151 -o /dev/null -w "%{speed_download}" "$m/$MODEL" 2>/dev/null)
  sp=${sp%.*}
  echo "$m -> ${sp}B/s"
  if awk "BEGIN{exit !($sp > $BESTSP)}" 2>/dev/null; then BEST="$m"; BESTSP=$sp; fi
done
echo "最快镜像: $BEST (${BESTSP}B/s)"
if [ -n "$BEST" ]; then
  echo "--- 全量下载 ---"
  curl -sL -m 500 -o /tmp/para.tar.bz2 "$BEST/$MODEL"
  SZ=$(stat -c%s /tmp/para.tar.bz2 2>/dev/null || echo 0)
  echo "下载大小: $SZ (期望77920048)"
  if tar xjf /tmp/para.tar.bz2 -C /tmp 2>/dev/null; then
    cp /tmp/sherpa-onnx-paraformer-zh-small-2024-03-09/model.int8.onnx /root/asr_model/
    cp /tmp/sherpa-onnx-paraformer-zh-small-2024-03-09/tokens.txt /root/asr_model/
    rm -f /tmp/para.tar.bz2
  else
    echo "解压失败(包不完整)"
  fi
fi
