ADB=/data/data/com.termux/files/usr/bin/adb
timeout 15 $ADB connect 127.0.0.1:41227 2>&1 | tail -1
timeout 15 $ADB connect 192.168.50.193:41227 2>&1 | tail -1
SER=$(timeout 10 $ADB devices 2>/dev/null | grep -m1 "device$" | awk '{print $1}')
if [ -n "$SER" ]; then
  echo "==adb已连 $SER=="
  echo "==Termux加入doze白名单=="
  $ADB -s $SER shell dumpsys deviceidle whitelist +com.termux 2>&1 | tail -2
  echo "==禁用doze深度休眠=="
  $ADB -s $SER shell dumpsys deviceidle disable 2>&1 | tail -2
  echo "==复查doze状态=="
  $ADB -s $SER shell dumpsys deviceidle 2>/dev/null | head -4
else
  echo "adb仍未连上"
fi
date "+%H:%M:%S"
