import ast, asyncio, os, sys
sys.path.insert(0,'/root/bot/bot')
from dotenv import load_dotenv
load_dotenv('/root/bot/bot/.env', override=True)
from persona import AZAR_PERSONA
src=open('/root/bot/bot/plugins/ai_chat.py').read()
root=ast.parse(src)
V={}
for node in root.body:
    if isinstance(node,ast.Assign):
        for t in node.targets:
            if isinstance(t,ast.Name):
                try: V[t.id]=ast.literal_eval(node.value)
                except: pass
def ev(n):
    if isinstance(n,ast.Name):
        if n.id=='AZAR_PERSONA': return AZAR_PERSONA
        return V.get(n.id,'')
    if isinstance(n,ast.Constant): return n.value
    if isinstance(n,ast.BinOp) and isinstance(n.op,ast.Add): return ev(n.left)+ev(n.right)
    try: return ast.literal_eval(n)
    except: return ''
SYS=''
for node in root.body:
    if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='SYSTEM_PROMPT' for t in node.targets):
        SYS=ev(node.value)

# 固定降级模型，测备忘放不同位置
CASES=[
 ("备忘在system_问好", NOTE_SYS, False, "@阿萨尔 在吗"),
 ("备忘在user末_问好", NOTE_USER, True, "@阿萨尔 在吗"),
 ("备忘在user末_油价", NOTE_USER, True, "@阿萨尔 油价又涨了"),
 ("无备忘_问好", None, False, "@阿萨尔 在吗"),
]
async def go():
    c=AsyncOpenAI(api_key=key,base_url=base,max_retries=0,timeout=25.0)
    try:
        for tag,note,in_user,content in CASES:
            print("=== CASE",tag,"===")
            msgs=[{"role":"system","content":SYS}]
            if note and not in_user: msgs.append({"role":"system","content":note})
            if note and in_user: content=content+"\n\n"+note

            for mdl in ['glm-4.5-flash','glm-4-flash']:
                try:
                    r=await c.chat.completions.create(
                        model=mdl,messages=msgs,max_tokens=200,temperature=0.7,
                        extra_body={"thinking":{"type":"disabled"}},
                    )

                    break
                except Exception as e: print("SKIP",mdl,type(e).__name__,str(e)[:80])
    finally: await c.close()
asyncio.run(go())
