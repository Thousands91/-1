import sherpa_onnx, subprocess, wave, array
r = sherpa_onnx.OfflineRecognizer.from_paraformer('/root/asr_model/model.int8.onnx','/root/asr_model/tokens.txt',num_threads=2,sample_rate=16000,feature_dim=80)
print('模型加载OK')
subprocess.run(['ffmpeg','-y','-i','/tmp/t.wav','-ar','16000','-ac','1','-f','wav','/tmp/t16.wav'],capture_output=True)
with wave.open('/tmp/t16.wav','rb') as w:
    frames = w.readframes(w.getnframes())
samples = list(array.array('h', frames))
s = r.create_stream()
s.accept_waveform(16000, [x/32768.0 for x in samples])
r.decode_stream(s)
print('转写管线OK, 输出:', repr(s.result.text))
