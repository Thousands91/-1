echo "==任务287712 安全重启bot加载新代码=="
OLD=""
for p in /proc/[0-9]*; do
  if [ "$(cat $p/comm 2>/dev/null)" = "python" ] && grep -qa "main.py" "$p/cmdline" 2>/dev/null; then
    OLD="$OLD ${p#/proc/}"
  fi
done
echo "旧bot进程:$OLD"
for pid in $OLD; do kill $pid 2>/dev/null; done
sleep 6
for pid in $OLD; do [ -d /proc/$pid ] && { echo "$pid还活着,补kill -9"; kill -9 $pid 2>/dev/null; }; done
sleep 3
NEW=""
for p in /proc/[0-9]*; do
  if [ "$(cat $p/comm 2>/dev/null)" = "python" ] && grep -qa "main.py" "$p/cmdline" 2>/dev/null; then
    NEW="$NEW ${p#/proc/}"
  fi
done
echo "杀后现状:$NEW"
if [ -z "$NEW" ]; then
  cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py >> /root/bot.log 2>&1 &
  echo "已按标准方式拉起"
fi
sleep 15
FINAL=""
for p in /proc/[0-9]*; do
  if [ "$(cat $p/comm 2>/dev/null)" = "python" ] && grep -qa "main.py" "$p/cmdline" 2>/dev/null; then
    FINAL="$FINAL ${p#/proc/}"
  fi
done
echo "最终bot进程:$FINAL (应为1个)"
echo "--- bot.log尾部 ---"
tail -12 /root/bot.log
