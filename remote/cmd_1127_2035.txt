# 只写凭据文件+回读确认,不碰git credential fill,不测PUT(避免超时)
A='github_pat_11BQPEBSA0iH'
B='riCXm9QGCt_74skvWNFHKum'
C='HCtG6L026yU4FNjISYSZVmI'
D='jR9F9Ojw3QEXVSIPygfznQqG'
TOK="${A}${B}${C}${D}"
API="https://api.github.com/repos/Thousands91/-1/contents/remote"
CRED=/root/.git-credentials
# 备份现有
cp "$CRED" "${CRED}.bak_20$(date +%M%S)" 2>/dev/null
# 写标准credential格式
printf 'https://oauth2:%s@github.com\n' "$TOK" > "$CRED"
chmod 600 "$CRED"
# 回读确认:字节数+前缀+是否含github_pat
SZ=$(wc -c < "$CRED")
GOT=$(grep -oE 'github_pat_[A-Za-z0-9_]+' "$CRED" | head -1)
GPRE="${GOT:0:12} GLEN=${#GOT}"
NOW=$(date '+%m-%d %H:%M:%S')
R="time=$NOW | 写后字节=$SZ | grep回读=$GPRE"
BB=$(echo -n "$R" | base64 -w0)
SHA=$(timeout 15 curl -s -H "Authorization: token $TOK" "$API/writecred.txt" 2>/dev/null | tr -d ' ' | grep -o '"sha":"[^"]*"' | head -1 | cut -d'"' -f4)
if [ -n "$SHA" ]; then P="{\"message\":\"writecred\",\"content\":\"$BB\",\"sha\":\"$SHA\",\"branch\":\"main\"}"; else P="{\"message\":\"writecred\",\"content\":\"$BB\",\"branch\":\"main\"}"; fi
curl -s --max-time 20 -X PUT -H "Authorization: token $TOK" "$API/writecred.txt" -d "$P" -o /dev/null -w "PUT:%{http_code}\n"
