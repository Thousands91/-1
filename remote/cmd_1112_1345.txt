echo "=== $(date '+%H:%M:%S') 补PIL+重启大脑 ==="
BOTD=$(dirname $(find /root -maxdepth 3 -name "main.py" -not -path "*/venv/*" 2>/dev/null | head -1)); echo "bot目录:$BOTD"
echo "--- 插件md5核对 ---"
md5sum $BOTD/plugins/fortune.py $BOTD/plugins/ai_chat.py 2>/dev/null
echo "--- 补装pillow ---"
pip3 install -q pillow 2>&1 | tail -1 || python3 -m pip install -q pillow 2>&1 | tail -1
/root/bot/venv/bin/python -c "import PIL; print('venv内PIL可用', PIL.__version__)"
echo "--- 重启大脑 ---"
pkill -f "venv/bin/python main.py"; sleep 2
cd "$BOTD" && nohup /root/bot/venv/bin/python main.py >> /root/bot.log 2>&1 &
sleep 12
tail -6 /root/bot.log
