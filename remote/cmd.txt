import urllib.request, urllib.parse, json, time
UA={"User-Agent":"curl/8.0","Accept":"application/json"}
def fetch(url,timeout=12):
    t=time.time()
    req=urllib.request.Request(url,headers=UA)
    raw=urllib.request.urlopen(req,timeout=timeout).read()
    return json.loads(raw), time.time()-t

# 1) 广州连续10次
url="https://wttr.in/"+urllib.parse.quote("广州")+"?format=j1&lang=zh"
ok=0; times=[]
for i in range(10):
    try:
        _,dt=fetch(url); ok+=1; times.append(f"{dt:.1f}s")
    except Exception as e:
        times.append(f"{type(e).__name__}")
print(f"[稳定] wttr.in 广州 10次: {ok}/10  [{', '.join(times)}]")

# 2) 中文多城市支持
