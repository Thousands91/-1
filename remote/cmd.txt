import urllib.request, json, time
cands = {
 "wttr.in j1": "https://wttr.in/Guangzhou?format=j1",
 "wttr.in 中文": "https://wttr.in/广州?format=j1&lang=zh",
 "open-meteo(备用域)": "https://api.open-meteo.com/v1/forecast?latitude=23.1&longitude=113.3&current=temperature_2m",
 "和风geo(无key看连通)": "https://geoapi.qweather.com/v2/city/lookup?location=广州&key=test",
}
for name,url in cands.items():
    t=time.time()
    try:
        req=urllib.request.Request(url,headers={"User-Agent":"curl/8.0","Accept":"application/json"})
        raw=urllib.request.urlopen(req,timeout=10).read()
        d=json.loads(raw)
        # 看关键字段
        keys=list(d.keys())[:6] if isinstance(d,dict) else type(d).__name__
        print(f"{name}: OK {len(raw)}B {time.time()-t:.1f}s keys={keys}")
        if 'wttr' in name and isinstance(d,dict):
            ca=d.get('current_condition',[{}])[0]
            print(f"   当前: temp={ca.get('temp_C')}C 天气={ca.get('lang_zh',[{}])[0].get('value') if ca.get('lang_zh') else ca.get('weatherDesc',[{}])[0].get('value')} 湿度={ca.get('humidity')}% 风={ca.get('windspeedKmph')}km/h")
            print(f"   未来天数: {len(d.get('weather',[]))}")
    except Exception as e:
        print(f"{name}: FAIL {type(e).__name__}:{str(e)[:50]} {time.time()-t:.1f}s")
PYEOF
echo "=====完====="
