import sys, ast, asyncio, os
from dotenv import load_dotenv
load_dotenv('/root/bot/bot/.env', override=True)
src=open('/root/bot/bot/plugins/ai_chat.py').read()
SHORT_SYS=None
for node in ast.parse(src).body:
    if isinstance(node,ast.Assign):
        for t in node.targets:
            if isinstance(t,ast.Name) and t.id=='SHORT_FOLLOWUP_SYSTEM': SHORT_SYS=ast.literal_eval(node.value)
def clean(v): return v.strip().strip('"').strip("'") if v else ''
key=clean(os.getenv('OPENAI_API_KEY') or os.getenv('SILICONFLOW_API_KEY'))
base=clean(os.getenv('OPENAI_BASE_URL')) or 'https://api.siliconflow.cn/v1'
msgs=[{"role":"user","content":"昨天那本书我翻到第三章了"},{"role":"assistant","content":"哦？讲到什么了，先生。"},{"role":"user","content":"？"}]
from openai import AsyncOpenAI
async def go():
    c=AsyncOpenAI(api_key=key,base_url=base,max_retries=0,timeout=20.0)
    try:
        for mdl in ['glm-4.5-flash','glm-4-flash','Qwen/Qwen2.5-7B-Instruct']:
            try:
                r=await c.chat.completions.create(model=mdl,messages=[{"role":"system","content":SHORT_SYS}]+msgs,max_tokens=120,temperature=0.7)
                print("MODEL_USED=",mdl)
                print("ANSWER=",(r.choices[0].message.content or '')[:250])
                break
            except Exception as e:
                print("SKIP",mdl,type(e).__name__)
    finally: await c.close()
asyncio.run(go())
