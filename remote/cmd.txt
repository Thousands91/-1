"""坑107取证插件:启动后翻群历史,抓正版音乐卡片的原始JSON做对照。
任务完成即自删(由运维远程清理)。"""
import asyncio, json
from nonebot import get_bot, get_driver
from nonebot.log import logger

async def dump():
    await asyncio.sleep(8)
    try:
        bot = get_bot()
    except Exception as e:
        logger.warning(f"[翻卡取证] get_bot失败: {e}")
        return
    found = 0
    out = open("/tmp/real_card.json", "w")
    for gid in (1109952783, 590454333):
        try:
            data = await bot.call_api("get_group_msg_history", group_id=gid, count=400)
            msgs = (data or {}).get("messages") or []
            logger.info(f"[翻卡取证] 群{gid} 拉到{len(msgs)}条")
            for m in msgs:
                raw = json.dumps(m, ensure_ascii=False)
                if "structmsg" in raw or '"ark"' in raw:
                    found += 1
                    out.write(raw + "\n")
        except Exception as e:
            logger.warning(f"[翻卡取证] 群{gid}失败: {type(e).__name__}: {e}")
    out.close()
    logger.info(f"[翻卡取证] 完成, {found} 条卡片消息存于 /tmp/real_card.json")

@get_driver().on_bot_connect
async def _(bot):
    asyncio.create_task(dump())
PYEOF
