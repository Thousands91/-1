echo "===== 真实heal进程(排除任务自匹配:只看cmdline精确等于bash+脚本路径的) ====="
N=0
for d in /proc/[0-9]*; do
  p=${d#/proc/}
  cl=$(tr '\0' ' ' < "$d/cmdline" 2>/dev/null)
  case "$cl" in
    "bash /root/azar_session_heal.sh "*) echo "  PID=$p CMD=[$cl]"; N=$((N+1)) ;;
  esac
done
echo "  真实进程数=$N (期望1)"
echo "===== PID文件 ====="; cat /root/.session_heal.lock.pid 2>/dev/null
echo "===== RUNDIR的hour(报错是否还在持续?看修改时间) ====="; ls -la --time-style=+%H:%M:%S /root/.session_heal/ 2>/dev/null
echo "===== 最近1分钟日志是否还在报错 ====="; tail -1 /root/session_heal.log 2>/dev/null; stat -c '日志修改:%y' /root/session_heal.log 2>/dev/null
echo "=====完====="
