import asyncio, json, re, sys
sys.path.insert(0, '/root/bot')
from nonebot import get_driver, get_bot
import nonebot

async def main():
    driver = get_driver()
    nonebot.init()
    # 直接用运行中的连接不行,这里起独立nonebot再连接也不行——改走反向:直接读NapCat配置里的反向ws地址,当客户端连大脑的ws
    # 更简单:从.env拿端口和token,直连大脑ws server
    conf = {}
    for line in open('/root/bot/bot/.env'):
        line=line.strip()
        if '=' in line and not line.startswith('#'):
            k,v=line.split('=',1); conf[k.strip()]=v.strip().strip('"')
    port = conf.get('PORT') or conf.get('ONEBOT_PORT') or '8080'
    tok  = conf.get('ONEBOT_ACCESS_TOKEN') or conf.get('ACCESS_TOKEN') or ''
    url = f"ws://127.0.0.1:{port}/onebot/v11/ws"
    import aiohttp
    hdr = {"Authorization": f"Bearer {tok}"} if tok else {}
    async with aiohttp.ClientSession() as s:
        try:
            ws = await s.ws_connect(url, headers=hdr, timeout=aiohttp.ClientWSTimeout(ws_close=10))
        except Exception as e:
            print("连接失败:", url, e); return
        for gid in (1109952783, 590454333):
            await ws.send_str(json.dumps({"action":"get_group_msg_history","params":{"group_id":gid,"count":300},"echo":f"h{gid}"}))
        found = 0
        end = 0
        while end < 2 and found < 3:
            try:
                m = await ws.receive(20)
            except Exception:
                break
            if m.type.name != 'TEXT': continue
            d = json.loads(m.data)
            if str(d.get("echo","")).startswith("h") and d.get("data"):
                end += 1
                for msg in (d["data"].get("messages") or []):
                    raw = json.dumps(msg, ensure_ascii=False)
                    if "structmsg" in raw or '"ark"' in raw or '音乐卡片' in raw or 'music' in raw.lower() and 'app' in raw:
                        found += 1
                        print(f"===群{gid} 发现卡片消息 seq={msg.get('message_seq')} time={msg.get('time')}===")
                        print(raw[:2500])
                        if found >= 3: break
        print(f"共找到 {found} 条卡片消息")
asyncio.run(main())
