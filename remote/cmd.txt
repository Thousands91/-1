UA='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
RID=128405
echo "--- [1] 搜索接口 web_albumpic_short ---"
curl -s --max-time 12 -A "$UA" "http://search.kuwo.cn/r.s?all=%E5%A4%A7%E6%82%B2%E5%92%92&ft=music&itemset=web_2013&client=mp3&rformat=json&encoding=utf8&pn=0&rn=1" 2>/dev/null | grep -oE 'web_albumpic_short[^&]*|PICPATH[^&]*|album[^&]*' | head -5
echo "--- [2] 歌曲详情接口 albumPic (多个候选) ---"
for U in \
  "http://www.kuwo.cn/api/v1/www/music/playInfo?mid=$RID" \
  "http://m.kuwo.cn/newh5/singles/songinfoandlrc?musicId=$RID" \
  "https://changenotice.kuwo.cn/mobi.s?rid=$RID&br=192"; do
  echo "接口:$U"
  curl -s --max-time 10 -A "$UA" "$U" 2>/dev/null | grep -oE '"albumPic":"[^"]*"|"pic":"[^"]*"|"picUrl":"[^"]*"|"albumpic":"[^"]*"|"songpic":"[^"]*"' | head -3
done
echo "--- [3] 直接看详情接口原文前400 ---"
curl -s --max-time 10 -A "$UA" "http://m.kuwo.cn/newh5/singles/songinfoandlrc?musicId=$RID" 2>/dev/null | head -c 400
