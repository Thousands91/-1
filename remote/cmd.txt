echo "=== 当前出口IP ==="
IPNOW=$(timeout 10 curl -s --max-time 8 https://api.ipify.org || timeout 10 curl -s --max-time 8 http://members.3322.org/dyndns/getip || timeout 10 curl -s --max-time 8 https://ifconfig.me)
echo "$IPNOW"
echo "=== 部署IP记录仪(每小时一行) ==="
cat > /root/ipwatch.sh << 'IPW'
#!/bin/bash
# 出口IP记录仪:每小时记一行,掉线排查时可回溯IP变动
while true; do
  IP=$(timeout 10 curl -s --max-time 8 https://api.ipify.org 2>/dev/null || timeout 10 curl -s --max-time 8 http://members.3322.org/dyndns/getip 2>/dev/null || echo "查询失败")
  echo "$(date '+%F %T') $IP" >> /root/ip.log
  sleep 3600
done
IPW
chmod +x /root/ipwatch.sh
setsid nohup /root/ipwatch.sh > /dev/null 2>&1 < /dev/null &
sleep 3
echo "=== 验证 ==="
pgrep -f ipwatch.sh >/dev/null && echo "IP记录仪在跑" || echo "没起来"
cat /root/ip.log 2>/dev/null
