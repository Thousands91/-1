echo "===== 坑159修复:引用图片不再被当动作 ====="
cd /root/bot/bot/plugins
cp ai_chat.py ai_chat.py.bak_quote159_$(date +%m%d%H%M)
python3 - <<'PYEOF'
p="ai_chat.py"
s=open(p,encoding="utf-8").read()
old='''                            _qparts.append(f"[表情包:{_qn}]")
                        else:
                            _qparts.append("[表情包]")'''
new='''                            # 坑159：引用的图片是"被引用的画面内容"，不是用户当面做的动作。
                            # 绝不能用 [表情包:] 标记——会被下方表情强提示的 re.sub 统一改成
                            # [动作:xxx]，让模型误读成"对方做了个叫'炸鸡配红酒'的动作"，
                            # 答非所问。改用自然语言表述画面，让模型针对图片内容回应。
                            _qparts.append(f"（引用一张图片，画面：{_qn}）")
                        else:
                            _qparts.append("（引用一张图片）")'''
assert s.count(old)==1, "锚点计数!=1,拒绝替换 count=%d"%s.count(old)
s=s.replace(old,new,1)
open(p,"w",encoding="utf-8").write(s)
print("替换成功")
PYEOF
echo "--- 语法检查 ---"
python3 -m py_compile ai_chat.py && echo "语法OK" || echo "语法FAIL-不重启"
echo "--- 新表述确认 ---"
grep -F -c '（引用一张图片，画面：{_qn}）' ai_chat.py
echo "--- 旧写法残留(应为0) ---"
grep -F -c '_qparts.append(f"[表情包:{_qn}]")' ai_chat.py
echo "--- md5 ---"
md5sum ai_chat.py
echo "=====完====="
