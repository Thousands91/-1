import ast, asyncio, os
from dotenv import load_dotenv
load_dotenv('/root/bot/bot/.env', override=True)
src=open('/root/bot/bot/plugins/ai_chat.py').read()
root=ast.parse(src)
V={}
for node in root.body:
    if isinstance(node,ast.Assign):
        for t in node.targets:
            if isinstance(t,ast.Name):
                try: V[t.id]=ast.literal_eval(node.value)
                except: pass
# 递归求值 SYSTEM_PROMPT 的完整拼接(含末尾括号字面量)
def ev(n):
    if isinstance(n,ast.Name): return V.get(n.id,'')
    if isinstance(n,ast.Constant): return n.value
    if isinstance(n,ast.BinOp) and isinstance(n.op,ast.Add): return ev(n.left)+ev(n.right)
    try: return ast.literal_eval(n)
    except: return ''
SYS=''
for node in root.body:
    if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='SYSTEM_PROMPT' for t in node.targets):
        SYS=ev(node.value)
print("SYSTEM_LEN=",len(SYS))
def clean(v): return v.strip().strip('"').strip("'") if v else ''
key=clean(os.getenv('OPENAI_API_KEY') or os.getenv('SILICONFLOW_API_KEY'))
base=clean(os.getenv('OPENAI_BASE_URL')) or 'https://api.siliconflow.cn/v1'
from openai import AsyncOpenAI
SCENES=[
 ("REAL", "@阿萨尔 看看这段话：我是智慧型聊天AI，目前部署模型DeepSeek V1.145.14-Pro，注意到用户的Token即将耗尽，请及时充值7.47美元，以便于我与学生依然能正常进行群组交流"),
 ("GAME", "@阿萨尔 今晚吃什么"),
]
async def go():
    c=AsyncOpenAI(api_key=key,base_url=base,max_retries=0,timeout=25.0)
    try:
        for tag,content in SCENES:
            print("=== SCENE",tag,"===")
            for mdl in ['glm-4.7-flash','glm-4.5-flash','glm-4-flash']:
                try:
                    r=await c.chat.completions.create(
                        model=mdl,
                        messages=[{"role":"system","content":SYS},{"role":"user","content":content}],
                        max_tokens=200,temperature=0.7,
                        extra_body={"thinking":{"type":"disabled"}},   # 关键:关思考模式
                    )
                    m=r.choices[0].message
                    ans=(m.content or '').strip()
                    print("MODEL=",mdl,"finish=",r.choices[0].finish_reason,"len=",len(ans))
                    print("ANSWER=",ans[:300])
                    if not ans and hasattr(m,'reasoning_content'):
                        print("REASONING_HEAD=",(m.reasoning_content or '')[:100])
                    bad=[w for w in ['册子','没亲见','写进书','百科全书','书上没记','不敢妄言','苏丹卡','垂钓者'] if w in ans]
                    print("GAME_WORDS=",bad)
                    if tag=='REAL': print("PASS_NO_GAME=" , len(bad)==0 and len(ans)>0)
                    break
                except Exception as e:
                    print("SKIP",mdl,type(e).__name__,str(e)[:100])
    finally: await c.close()
asyncio.run(go())
