import urllib.request, urllib.parse, json
UA={"User-Agent":"curl/8.0"}
d=json.loads(urllib.request.urlopen(urllib.request.Request("https://wttr.in/"+urllib.parse.quote("广州")+"?format=j1",headers=UA),timeout=12).read())
print("天数",len(d['weather']))
for i,day in enumerate(d['weather'][:3]):
    h=day['hourly'][0]
    codes=set(x.get('weatherCode') for x in day['hourly'])
    rains=[x.get('chanceofrain') for x in day['hourly']]
    print(f"第{i+1}天 date={day.get('date')} min={day.get('mintempC')} max={day.get('maxtempC')} codes={codes} 降雨概率={rains}")
PYEOF
echo "=====完====="
