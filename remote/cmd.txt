cat > /root/webui_state_watch.sh <<'WEBUI'
#!/bin/bash
# 可观测性改造1: 每轮记录WebUI完整登录状态到 webui_state.log
CFG=/root/Napcat/opt/QQ/resources/app/app_launcher/napcat/config/webui.json
PY=/root/bot/venv/bin/python3
[ -x "$PY" ] || PY=python3
T=$(grep -o '"token": "[^"]*"' "$CFG" 2>/dev/null | head -1 | sed 's/.*: "//;s/"//')
[ -n "$T" ] || exit 0
H=$($PY -c "import hashlib;print(hashlib.sha256(('$T'+'.napcat').encode()).hexdigest())" 2>/dev/null)
C=$(curl -s --max-time 5 -X POST http://127.0.0.1:6099/api/auth/login -H 'Content-Type: application/json' -d "{\"hash\":\"$H\"}" 2>/dev/null | $PY -c "import sys,json;print(json.load(sys.stdin)['data']['Credential'])" 2>/dev/null)
[ -n "$C" ] || exit 0
LINE=$(curl -s --max-time 6 -X POST http://127.0.0.1:6099/api/QQLogin/CheckLoginStatus -H "Authorization: Bearer $C" 2>/dev/null | $PY -c "
import sys,json,datetime
try:
 d=json.load(sys.stdin)['data']
 ts=datetime.datetime.now().strftime('%F %T')
 print(ts,'phase='+str(d.get('loginPhase')),'login='+str(d.get('isLogin')),'offline='+str(d.get('isOffline')),'core='+str(d.get('coreReady')),'qr='+str(d.get('qrLoginAccepted')),'err='+repr(d.get('loginError')))
except Exception as e: print(datetime.datetime.now().strftime('%F %T'),'ERR',str(e)[:60])
" 2>/dev/null)
[ -n "$LINE" ] && echo "$LINE" >> /root/webui_state.log
WEBUI
chmod +x /root/webui_state_watch.sh
echo "改造1脚本: $(md5sum /root/webui_state_watch.sh | cut -d' ' -f1)" >> "$LOG"

echo "--- login_state.log尾10 ---"; tail -10 /root/login_state.log 2>/dev/null
echo "--- webui_state.log尾10 ---"; tail -10 /root/webui_state.log 2>/dev/null
