HEALPAT='bash /root/azar_session_heal.[s]h'
echo "===== 两进程启动时间/父PID ====="
for p in $(pgrep -f "$HEALPAT"); do
  echo "PID $p PPid=$(awk '{print $4}' /proc/$p/stat 2>/dev/null) 启动=$(( $(date +%s) - $(stat -c %Y /proc/$p 2>/dev/null) ))秒前"
done
echo "===== 谁持flock锁(fuser) ====="
fuser /root/.session_heal.lock.flock 2>&1 | tr ' ' '\n' | grep -E '^[0-9]+$' | while read p; do echo "  持锁PID=$p cmd=$(tr '\0' ' ' </proc/$p/cmdline 2>/dev/null)"; done
echo "===== 完整新日志(最后8行带时间) ====="
tail -8 /root/session_heal.log 2>/dev/null
echo "===== 是否有别的watchdog在拉heal ====="
grep -rln 'session_heal' /root/*.sh /etc/cron* /var/spool/cron 2>/dev/null | head -10
echo "=====完====="
