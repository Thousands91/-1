echo "===== 探测密钥来源 $(date '+%T') ====="
PID=$(ps -eo pid,args | grep '[m]ain.py' | awk '{print $1}' | head -1)
echo "bot PID=$PID"
echo "--- 进程环境里有没有OPENAI(只看变量名和有否,不打印值) ---"
tr '\0' '\n' < /proc/$PID/environ 2>/dev/null | grep -oE '^[A-Z_]*(OPENAI|API|MODEL|KEY)[A-Z_]*=' | sort -u
echo "--- 有没有env文件 ---"
for f in /root/bot/.env /root/bot/bot/.env /root/.env /root/bot/venv/.env; do [ -f "$f" ] && echo "$f: $(grep -c OPENAI $f) 行OPENAI"; done
echo "--- 启动脚本里怎么注入的 ---"
grep -rnE 'OPENAI_API_KEY|export OPENAI|source .*env|set -a' /root/*.sh /root/bot/*.sh /root/.wsfix.sh 2>/dev/null | head -10
