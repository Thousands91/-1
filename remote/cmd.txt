import json, urllib.request, urllib.parse, html, time, difflib
env = {}
for line in open('/root/bot/bot/.env'):
    line = line.strip()
    if '=' in line and not line.startswith('#'):
        k, v = line.split('=', 1); env[k] = v

def ai(prompt, q):
    for i in range(4):
        body = json.dumps({'model': env.get('OPENAI_MODEL','glm-4.7-flash'),'max_tokens':60,'temperature':0,
         'thinking':{'type':'disabled'},'messages':[{'role':'system','content':prompt},{'role':'user','content':q}]}).encode()
        req = urllib.request.Request(env['OPENAI_BASE_URL']+'/chat/completions', data=body,
         headers={'Authorization':'Bearer '+env['OPENAI_API_KEY'],'Content-Type':'application/json'})
        try:
            r = json.load(urllib.request.urlopen(req, timeout=20))
            return r['choices'][0]['message']['content'].strip()
        except Exception as e:
            print(f'  AI第{i+1}次失败: {type(e).__name__}')
            time.sleep(3 + 3*i)
    return ''

def idx(p):
    arr = json.load(urllib.request.urlopen(urllib.request.Request(
        f'https://tbib.org/autocomplete.php?q={urllib.parse.quote(p)}', headers={'User-Agent':'Mozilla/5.0'}), timeout=15))
    out = []
    for it in arr or []:
        v = html.unescape(str(it.get('value',''))).strip()
        try: c = int(html.unescape(str(it.get('label',''))).rsplit('(',1)[-1].rstrip(') ').strip())
        except Exception: c = 0
        out.append((v, c))
    return out

def cnt(tag):
    d = urllib.request.urlopen(urllib.request.Request(
        f'https://tbib.org/index.php?page=dapi&s=post&q=index&limit=1&tags=rating:explicit+{urllib.parse.quote(tag)}',
        headers={'User-Agent':'Mozilla/5.0'}), timeout=15).read().decode()
    import re
    m = re.search(r'count=\"(\d+)\"', d)
    return int(m.group(1)) if m else 0

CAND = '你把中文搜索词翻译成二次元图库的英文检索词。只输出英文,最多3个不同说法的候选,用逗号分隔,全小写,用最朴素的常见英文单词或词组,不要自造复合词,不要解释。'
BLOCK = {'son','grandson','boy','child','kid','loli','shota','teen','young','brother'}
for word in ['奶奶','爷爷']:
    raw = ai(CAND, word)
    cands = [c.strip().strip(chr(34)).lower() for c in raw.split(',') if c.strip()]
    print(f'{word}: AI候选={cands}')
    tags = []
    for w in cands:
        w = w.replace(' ','_')
        if len(w) < 3: continue
        for n,c in idx(w):
            if n == w and c >= 3: tags.append(n); break
        else:
            best = {}
            prefix = w[:-1] if len(w) > 5 else w
            while len(prefix) >= 4:
                for n,c in idx(prefix):
                    if c < 3 or not n.startswith(prefix): continue
                    parts = n.split('_')
                    if w not in parts and not n.startswith(w+'_') and not n.endswith('_'+w): continue
                    if any(p in BLOCK for p in parts): continue
                    if difflib.SequenceMatcher(None, w, n).ratio() >= 0.6:
                        best[n] = max(best.get(n,0), c)
                prefix = prefix[:-1]
            for n,_ in sorted(best.items(), key=lambda kv:-kv[1])[:2]:
                if n not in tags: tags.append(n)
    for t in tags[:3]:
        print(f'  {t} explicit库存={cnt(t)}')
    time.sleep(2)
