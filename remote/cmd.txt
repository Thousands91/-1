#!/usr/bin/env python3
"""自托管音乐卡片签名服务 v2（坑107终修，2026-11-14）。

v1 教训：单线程 HTTPServer 一个慢请求（网易云接口抖动）堵死全部后续
请求，NapCat 超时拿空响应后走远程后备（死掉的 xingzhige）。
v2：多线程 + 请求日志 + 元数据缓存 + 短超时。
"""
import json
import random
import threading
import time
import urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

PORT = 10800
UIN = 3427943187
UA = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"}
_cache = {}  # sid -> (title, desc, pic, jump)
_lock = threading.Lock()


def _log(msg):
    with _lock:
        line = time.strftime("%m-%d %H:%M:%S ") + msg
        print(line, flush=True)


def build_ark(title, desc, jump, preview, tag, appid):
    return {
        "app": "com.tencent.structmsg",
        "config": {"ctime": 12, "forward": True, "type": "normal", "autosize": True},
        "extra": {"app_type": 1, "appid": appid,
                  "msg_seq": random.randint(100000000, 999999999), "uin": UIN},
        "meta": {"music": {
            "desc": desc or "分享",
            "jump": {"url": jump, "action": ""},
            "music": {"type": "song", "category_id": 100, "mid": "",
                      "passport": 2, "paste": 1},
            "preview": preview or "",
            "tag": tag,
            "title": title or "分享",
        }},
        "prompt": "[分享]" + (title or "音乐"),
    }


def netease_meta(sid):
    if sid in _cache:
        return _cache[sid]
    url = "https://music.163.com/api/song/detail/?id=%s&ids=%%5B%s%%5D" % (sid, sid)
    req = urllib.request.Request(url, headers=UA)
    d = json.load(urllib.request.urlopen(req, timeout=4))
    s = (d.get("songs") or [{}])[0]
    r = (s.get("name", ""),
         "/".join(a.get("name", "") for a in s.get("artists") or []),
         (s.get("album") or {}).get("picUrl", ""),
         "https://music.163.com/song?id=%s" % sid)
    _cache[sid] = r
    return r


class H(BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"

    def do_POST(self):
        n = int(self.headers.get("Content-Length") or 0)
        try:
            d = json.loads(self.rfile.read(n) or b"{}")
        except Exception:
            d = {}
        t = d.get("type", "?")
        out = None
        try:
            if t == "custom":
                out = build_ark(d.get("title", ""), d.get("singer") or d.get("content"),
                                d.get("url", ""), d.get("image", ""), "音乐", 100497308)
            elif t == "163":
                title, desc, pic, jump = netease_meta(str(d.get("id", "")))
                out = build_ark(title, desc, jump, pic, "网易云音乐", 100495022)
        except Exception as e:
            _log("build失败 type=%s id=%s err=%s" % (t, d.get("id", ""), e))
        if out is None:
            body = json.dumps({"error": "build failed", "ok": False}).encode()
            self.send_response(400)
        else:
            body = json.dumps(out, ensure_ascii=False).encode()
            self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)
        _log("已出卡 type=%s id=%s -> %d bytes" % (t, d.get("id", ""), len(body)))

    def log_message(self, *a):
        pass


_log("music_sign v2 启动 127.0.0.1:%d" % PORT)
ThreadingHTTPServer(("127.0.0.1", PORT), H).serve_forever()
