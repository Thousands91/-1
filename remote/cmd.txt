echo "=====1.当前bot进程====="
pgrep -f "venv/bin/python main.py" | head -3
echo "=====2.停旧进程（仅bot大脑，不碰NapCat）====="
pkill -f "venv/bin/python [m]ain.py"; sleep 3
pgrep -f "venv/bin/python main.py" >/dev/null && echo "仍有残留,强杀" && pkill -9 -f "venv/bin/python [m]ain.py" && sleep 2 || echo "旧进程已停"
echo "=====3.重新拉起====="
cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py > /root/bot.log 2>&1 < /dev/null &
sleep 18
echo "=====4.验证加载结果====="
NEWPID=$(pgrep -f "venv/bin/python main.py" | head -1)
echo "新PID=$NEWPID"
LOG=$(readlink /proc/$NEWPID/fd/1 2>/dev/null)
echo "日志=$LOG"
grep -a "Succeeded to load plugin\|Plugin .* loaded\|插件" $LOG 2>/dev/null | tail -15
echo "=====5.是否有报错====="
grep -a -iE "error|exception|traceback|failed" $LOG 2>/dev/null | tail -8 || echo "无报错"
echo "=====6.加载统计====="
grep -ac "Succeeded to load plugin" $LOG 2>/dev/null
echo "=====7.关键插件确认====="
grep -ao "plugins\.[a-z_]*" $LOG 2>/dev/null | sort -u | head -20
