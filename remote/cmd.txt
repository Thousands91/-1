set -e
TOKEN=$(grep -oE 'ghp_[A-Za-z0-9]+' /root/.git-credentials 2>/dev/null | head -1)
[ -n "$TOKEN" ] || { echo "FAIL:无token"; exit 1; }
echo "--- 1.下载两个插件 ---"
curl -s --max-time 60 -H "Authorization: token $TOKEN" "https://raw.githubusercontent.com/Thousands91/-1/main/code/bot/plugins/fortune.py" -o /root/fortune_new.py
curl -s --max-time 60 -H "Authorization: token $TOKEN" "https://raw.githubusercontent.com/Thousands91/-1/main/code/bot/plugins/ai_chat.py" -o /root/ai_chat_new.py
wc -c /root/fortune_new.py /root/ai_chat_new.py
echo "--- 2.语法检查 ---"
/root/bot/venv/bin/python -m py_compile /root/fortune_new.py /root/ai_chat_new.py && echo "语法通过" || { echo "语法失败"; exit 1; }
echo "--- 3.备份替换 ---"
cp /root/bot/bot/plugins/fortune.py /root/fortune.py.bak_1107
cp /root/bot/bot/plugins/ai_chat.py /root/ai_chat.py.bak_1107c
mv /root/fortune_new.py /root/bot/bot/plugins/fortune.py
mv /root/ai_chat_new.py /root/bot/bot/plugins/ai_chat.py
echo "--- 4.重启bot ---"
for p in $(pgrep -f "[m]ain.py"); do kill $p; done
echo "DEPLOY-OK"
