UA='Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
echo "===== 网易云标签名源实测 $(date '+%F %T') ====="
echo "-- 1. xingzhige 带 appid 覆盖参数 --"
for payload in \
  '{"server":"netease","id":5262296,"type":"163","appid":100951776}' \
  '{"server":"netease","id":5262296,"type":"163","app":"com.netease.cloudmusic"}' \
  '{"server":"netease","id":5262296,"type":"netease"}'; do
  R=$(curl -s --max-time 12 -A "$UA" -X POST "https://ss.xingzhige.com/music_card/card" -H "Content-Type: application/json" -d "$payload" 2>/dev/null)
  echo "[$payload] -> $(echo "$R" | grep -o '"app":"[^"]*"\|"appid":[0-9]*' | head -4 | tr '\n' ' ')"
done
