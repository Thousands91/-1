echo "===== 287440 清多余autoheal(只读验收同任务) ====="
echo "--- 当前实例 ---"
ps -eo pid,args | grep -E "remote_ctl|azar_autoheal|offline_watch" | grep -v grep

KEEP=$(echo "$PIDS" | head -1)

done
