echo "=====1.重启守护使新参数生效====="
OLDPID=$(pgrep -f "bash /root/azar_guard[.]sh" | head -1)
echo "旧守护PID=$OLDPID"
[ -n "$OLDPID" ] && kill "$OLDPID" 2>/dev/null && echo "已停止旧守护"
sleep 2
setsid nohup bash /root/azar_guard.sh > /dev/null 2>&1 < /dev/null &
sleep 5
NEWPID=$(pgrep -f "bash /root/azar_guard[.]sh" | head -1)
echo "新守护PID=$NEWPID"
echo "=====2.确认守护运行中====="
[ -n "$NEWPID" ] && kill -0 "$NEWPID" 2>/dev/null && echo "守护运行中" || echo "守护未启动!"
echo "=====3.守护日志确认====="
tail -5 /root/guard_status.txt 2>/dev/null
