PID=$(pgrep -f 'venv/bin/python main.py' | head -1)
echo "大脑PID: $PID"
echo "=== 大脑进程环境变量 ==="
cat /proc/$PID/environ | tr '\0' '\n' | grep -E "^OPENAI_|^ZHIPU_|^SILICONFLOW" | sed 's/\(=.\{12\}\).*/\1***/'
echo "=== .env 原文（含重复键检查） ==="
grep -n "OPENAI\|ZHIPU\|SILICONFLOW" /root/bot/bot/.env | sed 's/\(=.\{12\}\).*/\1***/'
echo "=== .env.prod / .env.other 检查 ==="
ls /root/bot/bot/.env* 2>/dev/null
for f in /root/bot/bot/.env.*; do [ -f "$f" ] && echo "--$f--" && grep -E "OPENAI|ZHIPU" "$f" | sed 's/\(=.\{12\}\).*/\1***/'; done
