echo "===== 扫码时刻 vs 会话存活时长取证(只读) ====="
LS=/root/login_state.log
echo "--- 文件信息 ---"
ls -la "$LS" 2>/dev/null; echo "总行数: $(wc -l < "$LS" 2>/dev/null)"
echo "--- 状态变化全记录(去掉每5分钟重复采样,只留变化点) ---"
awk '
{
  # 行格式猜测: 时间 状态, 先原样打印前20行确认
  if (NR<=20) print NR": "$0
}' "$LS" 2>/dev/null
echo "--- 采样样例(头10尾10) ---"
head -10 "$LS" 2>/dev/null; echo "..."; tail -10 "$LS" 2>/dev/null
