import datetime, re

# 1. tick.log 全量断档(>120秒)
ticks = []
for ln in open('/root/tick.log', errors='ignore'):
    m = re.match(r'(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})', ln)
    if m:
        try: ticks.append(datetime.datetime.strptime(m.group(1), '%Y-%m-%d %H:%M:%S'))
        except: pass
ticks.sort()
print("[1]tick.log总样本:", len(ticks), "覆盖:", ticks[0], "->", ticks[-1] if ticks else "")
gaps = []
for i in range(1, len(ticks)):
    d = (ticks[i] - ticks[i-1]).total_seconds()
    if d > 120:
        gaps.append((ticks[i-1], ticks[i], int(d)))
print("[2]冻结断档(>2分钟)共%d个:" % len(gaps))
byday = {}
for a, b, d in gaps:
    k = a.strftime('%m-%d')
    byday.setdefault(k, []).append((a, b, d))
for k in sorted(byday):
    day = byday[k]
    total = sum(d for _,_,d in day)
    mx = max(day, key=lambda x:x[2])
    print(f"  {k}: 冻结{len(day)}次 总{total//60}分 最长{mx[2]//60}分(单次 {mx[0].strftime('%H:%M')}->{mx[1].strftime('%H:%M')})")
# 11-19与11-20的断档明细
for k in ('11-19','11-20'):
    if k in byday:
        print(f"[3]{k}断档明细(>10分钟的):")
        for a,b,d in byday[k]:
            if d > 600:
                print(f"  {a.strftime('%H:%M')} -> {b.strftime('%H:%M')}  {d//60}分")

# 2. login_state 状态变化时刻
print("[4]login_state最近状态变化:")
for ln in open('/root/login_state.log', errors='ignore'):
    if '状态变化' in ln: print(" ", ln.strip()[-80:])
