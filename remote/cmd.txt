import urllib.request, urllib.parse, json, time
UA={"User-Agent":"curl/8.0","Accept":"application/json"}
d=json.loads(urllib.request.urlopen(urllib.request.Request("https://wttr.in/"+urllib.parse.quote("广州")+"?format=j1&lang=zh",headers=UA),timeout=12).read())
ca=d['current_condition'][0]
print("[当前] keys:",list(ca.keys()))
for k in ['temp_C','FeelsLikeC','humidity','windspeedKmph','winddir16Point','weatherCode','weatherDesc','lang_zh']:
    print(f"   {k} = {ca.get(k)}")
w=d['weather'][0]
print("[日] keys:",list(w.keys()))
for k in ['date','mintempC','maxtempC','avgtempC']:
    print(f"   {k} = {w.get(k)}")
h=w['hourly'][0]
print("[时] keys:",list(h.keys()))
for k in ['weatherCode','weatherDesc','lang_zh','chanceofrain']:
    print(f"   {k} = {h.get(k)}")
print("[未来天数]",len(d.get('weather',[])))
# 全部3天的日期/温度/是否有lang_zh
for i,day in enumerate(d.get('weather',[])[:3]):
    hh=day['hourly'][0]
    print(f"   第{i+1}天 date={day.get('date')} min={day.get('mintempC')} max={day.get('maxtempC')} code={hh.get('weatherCode')} lang_zh={hh.get('lang_zh')} rain={hh.get('chanceofrain')}")
PYEOF

import urllib.request, time
UA={"User-Agent":"curl/8.0"}
u="https://api.open-meteo.com/v1/forecast?latitude=23.1&longitude=113.3&current=temperature_2m"
for i in range(3):
    t=time.time()
    try:
        urllib.request.urlopen(urllib.request.Request(u,headers=UA),timeout=10).read()
        print(f"   #{i+1} OK {time.time()-t:.1f}s")
    except Exception as e:
        print(f"   #{i+1} FAIL {type(e).__name__} {time.time()-t:.1f}s")
