import asyncio, json, os, websockets
TOK=os.getenv("ONEBOT_ACCESS_TOKEN","")
print("token存在:",bool(TOK))
async def main():
    uri="ws://127.0.0.1:8080/onebot/v11/ws"+(("?access_token="+TOK) if TOK else "")
    try:
        ws=await asyncio.wait_for(websockets.connect(uri,extra_headers={"Authorization":"Bearer "+TOK} if TOK else None),timeout=10)
    except Exception as e:
        print("连接失败:",type(e).__name__,str(e)[:120]); return
    try:
        await ws.send(json.dumps({"action":"get_msg","params":{"message_id":1780830792},"echo":"g1"}))
        for _ in range(12):
            try:
                raw=await asyncio.wait_for(ws.recv(),timeout=8)
            except asyncio.TimeoutError:
                print("收超时"); break
            d=json.loads(raw)
            if d.get("echo")=="g1":
                msg=(d.get("data") or {}).get("message") or []
                print("消息段数:",len(msg) if isinstance(msg,list) else "非list")
                if isinstance(msg,list):
                    for seg in msg:
                        if isinstance(seg,dict) and seg.get("type")=="image":
                            print("IMAGE_URL_FULL:",(seg.get("data") or {}).get("url",""))
                break
    finally:
        await ws.close()
asyncio.run(main())
PYEOF
