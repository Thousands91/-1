echo "===== 找bot的运行python ====="
ps aux | grep -E 'main.py|nonebot|python' | grep -v grep | head -10
echo "--- venv候选 ---"
ls -d /root/bot/venv /root/bot/.venv /root/venv 2>/dev/null
echo "--- 哪个python有nonebot ---"
for p in python3 /root/bot/venv/bin/python /root/bot/.venv/bin/python; do
  [ -x "$p" ] || command -v "$p" >/dev/null 2>&1 || continue
  echo -n "$p: "; "$p" -c "import nonebot; print('有nonebot')" 2>/dev/null || echo "无"
done
