  echo "用bot目录版启动"

ls /proc/*/cmdline 2>/dev/null | head -1 >/dev/null && for p in /proc/[0-9]*/cmdline; do tr '\0' ' ' < "$p" 2>/dev/null | grep -qi napcat && echo "活: $(cat /proc/${p//[!0-9]/}/cmdline 2>/dev/null | tr '\0' ' ' | head -c 80)"; done | head -3
