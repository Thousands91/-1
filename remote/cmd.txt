echo "=====2.每次崩溃前的错误类型====="
grep -a "error_code" /root/napcat-restart.log | grep -o "error_code=[0-9]*" | sort | uniq -c | sort -rn | head
echo "=====3.崩溃时刻（FATAL行的时间戳）====="
grep -a "FATAL" /root/napcat-restart.log | grep -o "[0-9]\{4\}/[0-9]\{6\}" | cut -c6-7 | sort | uniq -c
echo "=====4.当前启动命令====="
ps aux | grep "[o]pt/QQ/qq" | head -2
echo "=====5.Xvfb进程数====="
ps aux | grep -c "[X]vfb"
echo "=====6.GPU相关环境变量====="
PID=$(pgrep -f "opt/QQ/qq" | head -1); tr "\0" "\n" < /proc/$PID/environ 2>/dev/null | grep -i -E "gpu|gl|render|display|angle" | head -10

grep -a -c -E "踢|下线|失效|被禁止|logout" /root/napcat-restart.log
