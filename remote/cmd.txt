from datetime import datetime
ts=[]
for ln in open('/root/tick.log',errors='replace'):
    ln=ln.strip()
    try: ts.append(datetime.strptime(ln[:19],'%Y-%m-%d %H:%M:%S'))
    except: pass
gaps=[]
for i in range(1,len(ts)):
    d=(ts[i]-ts[i-1]).total_seconds()
    if d>300: gaps.append((ts[i-1],ts[i],d))
drops=[]
for ln in open('/root/login_state.log',errors='replace'):
    if '状态变化: [在线] -> [未登录]' in ln:
        try: drops.append(datetime.strptime(ln[:19],'%Y-%m-%d %H:%M:%S'))
        except: pass
print('[DROP] 全部13次, 每次掉线前6h最大tick断档:')
for dp in drops:
    pre=[d for a,b,d in gaps if 0 < (dp-b).total_seconds() <= 6*3600]
    mx=max(pre) if pre else 0
    seg='白天' if 6<=dp.hour<18 else '夜间'
    print(f'{dp.strftime("%m-%d %H:%M:%S")} {seg} 前6h断档={int(mx)}s({mx/60:.0f}m)')

print('[TODAY] 11-22 00:00之后 tick 全部断档:')
