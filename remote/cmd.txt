HEALPAT='bash /root/azar_session_heal.[s]h'
# 1) 加入开机自启(方括号隔离,仿autoheal行)
if ! grep -q 'azar_session_heal.sh' /root/start_all.sh; then
  echo "pgrep -f 'bash /root/azar_session_heal.[s]h' >/dev/null || ( setsid nohup bash /root/azar_session_heal.sh >> /root/session_heal.log 2>&1 < /dev/null & )" >> /root/start_all.sh
  echo "[1] 已加入start_all.sh自启"
else
  echo "[1] start_all.sh已有该行,跳过"
fi
echo "[2] 自启行:"; grep -n 'session_heal' /root/start_all.sh
# 2) 立即启动
pgrep -f "$HEALPAT" >/dev/null || ( setsid nohup bash /root/azar_session_heal.sh >> /root/session_heal.log 2>&1 < /dev/null & )
sleep 3
echo "[3] 启动后进程:"
ps -ef | grep '[s]ession_heal' | head -3
echo "[4] 启动日志:"; tail -3 /root/session_heal.log 2>/dev/null
