export PATH=/data/data/com.termux/files/usr/bin:$PATH
export LD_LIBRARY_PATH=/data/data/com.termux/files/usr/lib
export HOME=/data/data/com.termux/files/home
export PREFIX=/data/data/com.termux/files/usr
echo "=== 试apt直装 ==="
yes | apt install -y android-tools 2>&1 | grep -E "Setting up|E:|Err|root" | tail -3
if [ ! -x $PREFIX/bin/adb ]; then
  echo "=== apt被拒,手动deb解包 ==="
  M=$(curl -sL "https://packages.termux.dev/apt/termux-main/dists/stable/main/binary-aarch64/" >/dev/null; curl -sL "https://packages.termux.dev/apt/termux-main/dists/stable/main/binary-aarch64/Packages" | grep -A5 "^Package: android-tools$" | grep "^Filename:" | head -1 | cut -d' ' -f2)
  echo "deb路径: $M"
  curl -sL -o /tmp/at.deb "https://packages.termux.dev/apt/termux-main/$M" && dpkg -x /tmp/at.deb /tmp/atx && echo 解包OK
  ls /tmp/atx/data/data/com.termux/files/usr/bin/ 2>/dev/null | head -8
  cp -r /tmp/atx/data/data/com.termux/files/usr/bin/* $PREFIX/bin/ 2>/dev/null
  cp -r /tmp/atx/data/data/com.termux/files/usr/lib/* $PREFIX/lib/ 2>/dev/null
fi
echo "=== adb版本与pair确认 ==="
$PREFIX/bin/adb version 2>&1 | head -1
$PREFIX/bin/adb help 2>&1 | grep -E "pair|connect" | head -4
