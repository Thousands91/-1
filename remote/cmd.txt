echo "===== 网易云标API最后探查 $(date '+%F %T') ====="
echo "-- 1. vkeys 音乐卡接口 --"
for u in \
  'https://api.vkeys.cn/v2/music/?src=netease&id=5262296' \
  'https://api.vkeys.cn/v3/music/?src=netease&id=5262296' \
  'https://api.vkeys.cn/v2/music/card?server=netease&id=5262296'; do
  echo "[$u] -> $(curl -s --max-time 10 -A "$UA" "$u" 2>/dev/null | head -c 200)"
  echo ""
done
echo "-- 2. 抓一张xingzhige完整卡json存文件(供分析) --"
curl -s --max-time 12 -A "$UA" -X POST 'https://ss.xingzhige.com/music_card/card' -H 'Content-Type: application/json' -d '{"server":"netease","id":5262296,"type":"163"}' > /root/xz_card_sample.json 2>/dev/null
echo "字节:$(wc -c </root/xz_card_sample.json 2>/dev/null)"
grep -o '"app":"[^"]*"\|"appid":[0-9]*\|"token":"[^"]*"\|"jumpUrl":"[^"]*"' /root/xz_card_sample.json 2>/dev/null | head -8
