import sys, asyncio, types
# stub nonebot 相关模块,避免顶层 on_message() 报未初始化
def mkmod(name):
    m = types.ModuleType(name); sys.modules[name] = m; return m
nb = mkmod('nonebot')
class _Matcher:
    def __init__(self,*a,**k): pass
    def handle(self,*a,**k): return lambda f:f
nb.on_message = lambda *a,**k: _Matcher()
nb.on_notice = lambda *a,**k: _Matcher()
nb.on_command = lambda *a,**k: _Matcher()
nb.get_driver = lambda *a,**k: types.SimpleNamespace(on_startup=lambda f:f, on_shutdown=lambda f:f)
nb.get_plugin = lambda *a,**k: None
# 常见子模块
for sub in ['on_message','on_notice','on_command','get_driver','get_plugin','require','load_builtin_plugins']:
    setattr(nb, sub, (lambda *a,**k: (lambda f:f)()))
adapters = mkmod('nonebot.adapters')
onebot = mkmod('nonebot.adapters.onebot')
v11 = mkmod('nonebot.adapters.onebot.v11')
class _Event: pass
class _Bot: pass
class _Message(list):
    def __init__(self,*a,**k): super().__init__()
for mod in [adapters,onebot,v11]:
    mod.Event=_Event; mod.Bot=_Bot; mod.Message=_Message
onebot.v11=v11
mkmod('nonebot.rule'); mkmod('nonebot.typing'); mkmod('nonebot.exception')
try:
    sys.path.insert(0,'/root/bot/bot/plugins')
    import ai_chat as A

      {"role":"user","content":"昨天那本书我翻到第三章了"},
      {"role":"assistant","content":"哦？讲到什么了，先生。"},

    ]
    async def go():
        r=await A.call_llm(msgs, system_override=A.SHORT_FOLLOWUP_SYSTEM)
        print("ANSWER=",r[:250])
    asyncio.run(go())
except Exception as e:
    import traceback; traceback.print_exc()
