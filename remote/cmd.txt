import re, glob, os, datetime

# 1) 找候选 tick 文件：/root 下含 11-19 时间戳记录、按时间递增的小日志
cands = []
for p in glob.glob('/root/*') + glob.glob('/root/logs/*'):
    if not os.path.isfile(p):
        continue
    if os.path.getsize(p) > 30 * 1024 * 1024:
        continue
    try:
        with open(p, errors='replace') as f:
            f.seek(max(0, os.path.getsize(p) - 300 * 1024))
            tail = f.read()
    except Exception:
        continue
    if re.search(r'11-19 \d{2}:\d{2}:\d{2}', tail) or re.search(r'11-19T\d{2}:\d{2}', tail):
        cands.append((p, os.path.getsize(p)))
print('候选文件:', cands)

def parse_lines(path):
    rows = []
    with open(path, errors='replace') as f:
        for ln in f:
            m = re.search(r'(11-\d{2}) (\d{2}:\d{2}:\d{2})', ln)
            if m:
                hh, mm, ss = m.group(2).split(':')
                rows.append((m.group(1), int(hh)*3600+int(mm)*60+int(ss), ln.strip()[:80]))
            else:
                m2 = re.search(r'(11-\d{2})T(\d{2}:\d{2}:\d{2})', ln)
                if m2:
                    hh, mm, ss = m2.group(2).split(':')
                    rows.append((m2.group(1), int(hh)*3600+int(mm)*60+int(ss), ln.strip()[:80]))
    return rows

def grade(sec):
    if sec > 900: return '深冻'
    if sec > 300: return '半冻'
    if sec > 60:  return '浅冻'
    return '微冻'

for path, _ in cands:
    rows = parse_lines(path)
    d19 = [r for r in rows if r[0] == '11-19']
    if len(d19) < 20:
        continue
    print(f'>>> {path}  11-19记录 {len(d19)} 条')
    prev = None
    gaps = []
    for day, sec, ln in d19:
        if prev is not None and sec > prev[0]:
            g = sec - prev[0]
            if g > 55:
                gaps.append((prev[0], sec, g, prev[1]))
        prev = (sec, ln)
    if not gaps:
        print('  11-19 全天无 >55s 断档')
    for s, e, g, lastln in gaps:
        fs, fe = s % 86400, e % 86400
        def fmt(x):
            return f'{x//3600:02d}:{(x%3600)//60:02d}:{x%60:02d}'
        print(f'  [{grade(g)}] {fmt(fs)} -> {fmt(e%86400)}  断 {g}s  (断前最后一条: {lastln[:60]})')
    # 汇总：15:30 前后对比
    am = [g for s, e, g, _ in gaps if (s % 86400) < 15.5*3600]
    pm = [g for s, e, g, _ in gaps if (s % 86400) >= 15.5*3600]
    print(f'  15:30前 断档{len(am)}个 最大{max(am) if am else 0}s | 15:30后 断档{len(pm)}个 最大{max(pm) if pm else 0}s')
print('--- C 完 ---')
PYEOF
