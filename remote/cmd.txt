echo "===== [1] NapCat 相关日志文件(名称/大小/时间) ====="
ls -la /root/*.log /root/napcat*/**/*.log /root/Napcat/**/*.log 2>/dev/null | grep -iE 'napcat|qq|offline|forensic' | head -60
echo "===== [2] 所有 .bak 备份 ====="
find /root -maxdepth 4 -type f \( -name '*.bak*' -o -name '*bak_*' \) 2>/dev/null | head -60
echo "===== [3] 各日志中 KickedOffLine 计数 ====="
for f in $(find /root -maxdepth 4 -type f \( -name '*.log' -o -name '*.bak*' \) 2>/dev/null | grep -iE 'napcat|offline|qq' | head -40); do
  c=$(grep -c 'KickedOffLine' "$f" 2>/dev/null)
  [ "${c:-0}" -gt 0 ] && echo "$c  $f"
done
