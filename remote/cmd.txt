PREFIX=/data/data/com.termux/files/usr
U=https://mirrors.tuna.tsinghua.edu.cn/termux/apt/termux-main
IDX=/tmp/termux_packages.idx
echo "=== idx里搜utf8相关包 ==="
grep -B1 "utf8" $IDX | grep -E "^Package|^Filename" | head -6
F=$(grep -A8 "^Package: libprotobuf$" $IDX | grep "^Filename:" | head -1 | awk '{print $2}')
ENC=$(echo "$F" | sed 's/:/%3a/g')
curl -sL --max-time 120 -o /tmp/lp.deb "$U/$ENC" && echo "=== libprotobuf包完整文件清单 ===" && dpkg -c /tmp/lp.deb | awk '{print $NF}' | grep lib
dpkg -x /tmp/lp.deb /tmp/lpx 2>/dev/null
find /tmp/lpx -name "*.so*" -exec cp -a {} /root/adblibs/ \; 2>/dev/null
echo "=== adb终试 ==="
export LD_LIBRARY_PATH=/root/adblibs:$PREFIX/lib
/root/adblibs/adb version 2>&1 | head -2
/root/adblibs/adb help 2>&1 | grep -E "pair|connect" | head -4
