import sys, asyncio, types
def newmod(name, **attrs):
    m = types.ModuleType(name)
    for k,v in attrs.items(): setattr(m,k,v)
    sys.modules[name] = m
    return m
class _Matcher:
    def __init__(self,*a,**k): pass
    def handle(self,*a,**k): return lambda f:f
class _Event: pass
class _Bot: pass
class _Message(list):
    def __init__(self,*a,**k): super().__init__()
    def __add__(self,o): return self
class _Rule:
    def __and__(self,o): return self
    def __or__(self,o): return self
logger = types.SimpleNamespace(info=lambda *a,**k:None, warning=lambda *a,**k:None, error=lambda *a,**k:None, debug=lambda *a,**k:None)
nb = newmod('nonebot',

    get_driver=lambda *a,**k:types.SimpleNamespace(on_startup=lambda f:f, on_shutdown=lambda f:f, config=types.SimpleNamespace()),
    get_plugin=lambda *a,**k:None, require=lambda *a,**k:None,
    load_builtin_plugins=lambda *a,**k:None, init=lambda *a,**k:None,
)
