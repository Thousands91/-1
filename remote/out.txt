helper = '''
    def _read_body(self):
        """NapCat 的 HttpGetJson 发 POST 不带 Content-Length，body 走 chunked；
        Python 标准库只会按 Content-Length 读，结果读到 0 字节——这就是
        v2 时代 NapCat 三连敲门全收到 type=? 的原因。两种读法都支持。"""
        te = (self.headers.get("Transfer-Encoding") or "").lower()
        if "chunked" in te:
            body = b""
            while True:
                line = self.rfile.readline(65536).strip()
                if b";" in line:
                    line = line.split(b";")[0]
                try:
                    size = int(line, 16)
                except ValueError:
                    break
                if size == 0:
                    self.rfile.readline(65536)
                    break
                body += self.rfile.read(size)
                self.rfile.read(2)
            return body
        n = int(self.headers.get("Content-Length") or 0)
        return self.rfile.read(n) if n else b""

    def do_POST(self):'''
t = t.replace("    def do_POST(self):", helper, 1)
t = t.replace(old, new, 1)
open(p, "w", encoding="utf-8").write(t)
print("chunked 支持已写入")
