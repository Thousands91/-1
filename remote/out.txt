#!/bin/bash
# 冻结取证秒表:每60秒写一行时间戳,断档=冻结窗口
while true; do
  echo "$(date '+%F %T')" >> /root/tick.log
  sleep 60
done

chmod +x /root/tick.sh
setsid nohup /root/tick.sh > /dev/null 2>&1 < /dev/null &
sleep 3
