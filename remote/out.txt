import nonebot, asyncio, hashlib, time
nonebot.init()
from nonebot.adapters.onebot.v11 import Adapter
nonebot.get_driver().register_adapter(Adapter)
import plugins.ai_chat as ac
# 一张稳定的公开图(猫图),确保未在缓存
URL="https://httpbin.org/image/jpeg"
async def main():
    ac._IMG_CAP_LOADED=False; ac._load_img_caps()
    print("开始同步识图(wait=True)...")
    t=time.time()
    r=await ac._image_sticker_name(URL, wait=True)
    print("返回:",r,"耗时:%.1fs"%(time.time()-t))
    # 关键: 返回不应是占位"表情包";且缓存应写入了真名
    print("是否占位符:", r=="表情包")
    print("缓存写入:", {k[:8]:v for k,v in ac._IMG_CAPS.items()})
asyncio.run(main())
