--- 命令 ---
ADB=/data/data/com.termux/files/usr/bin/adb
echo "== 重置adb server =="
timeout 15 $ADB kill-server 2>&1
sleep 2
timeout 15 $ADB start-server 2>&1 | head -2
echo "== 双地址尝试连接 :38519 =="
timeout 12 $ADB connect 127.0.0.1:38519 2>&1
timeout 12 $ADB connect 192.168.31.28:38519 2>&1
echo "== 设备列表 =="
timeout 10 $ADB devices 2>&1
echo "== 若已连接则关闭杀手 =="
D=$(timeout 10 $ADB devices 2>/dev/null | grep -c "device$")
if [ "$D" -ge 1 ]; then
  timeout 12 $ADB shell settings put global settings_enable_monitor_phantom_procs false 2>&1
  timeout 12 $ADB shell device_config put activity_manager max_phantom_processes 2147483647 2>&1
  echo "== 验证 =="
  timeout 12 $ADB shell settings get global settings_enable_monitor_phantom_procs 2>&1
else
  echo 无可用设备连接
fi
date "+完成: %H:%M:%S"
