--- 命令 ---
T=$(grep -oE "ghp_[A-Za-z0-9]+" /root/.git-credentials | head -1)
ADB=/data/data/com.termux/files/usr/bin/adb
$ADB start-server >/dev/null 2>&1
OK=0
for i in 1 2 3 4 5 6 7 8; do
  PIN=$(curl -sL -H "Authorization: token $T" "https://raw.githubusercontent.com/Thousands91/-1/main/remote/pin.txt" | tr -d '\r\n')
  if [ -n "$PIN" ]; then
    ADDR=${PIN%% *}; CODE=${PIN##* }
    echo "配对信息到货: $ADDR (码${#CODE}位)"
    if echo "$CODE" | $ADB pair "$ADDR" 2>&1; then OK=1; break; fi
  fi
  sleep 4
done
echo "== 配对结果 OK=$OK =="
echo "== mdns服务发现 =="
$ADB mdns services 2>&1 | head -6
CPORT=$($ADB mdns services 2>/dev/null | grep "_adb-tls-connect" | grep -oE ":[0-9]+" | head -1 | tr -d ':')
if [ -n "$CPORT" ]; then echo "发现连接端口$CPORT,尝试直连"; $ADB connect 192.168.31.28:$CPORT 2>&1; fi
$ADB devices
