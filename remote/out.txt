echo "===== 真实heal进程(排除任务自匹配:只看cmdline精确等于bash+脚本路径的) ====="
N=0
for d in /proc/[0-9]*; do
  p=${d#/proc/}
  cl=$(tr '\0' ' ' < "$d/cmdline" 2>/dev/null)
  case "$cl" in
    "bash /root/azar_session_heal.sh "*) echo "  PID=$p CMD=[$cl]"; N=$((N+1)) ;;
  esac
done
echo "  真实进程数=$N (期望1)"
