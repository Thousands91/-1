--- 命令 ---
echo "===== 重启大脑加载酷我代码 $(date '+%F %T') ====="
echo "--- [1] 停旧大脑 ---"
OLDPID=$(pgrep -f '/root/bot/venv/bin/python main.py' | tr '\n' ' ')
echo "旧PID: $OLDPID"
pkill -f '/root/bot/venv/bin/python main.py' 2>/dev/null; sleep 3
pgrep -f '/root/bot/venv/bin/python main.py' | wc -l | xargs echo "停后剩余:"
echo "--- [2] setsid拉起新大脑 ---"
cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py >> /root/bot/bot/bot.log 2>&1 &
sleep 12
echo "--- [3] 进程确认 ---"
pgrep -af '/root/bot/venv/bin/python main.py' | head -3
echo -n "实例数: "; pgrep -f '/root/bot/venv/bin/python main.py' | wc -l
echo "--- [4] 启动日志确认(song_request加载+Bot连接) ---"
tail -60 /root/bot/bot/bot.log 2>/dev/null | grep -iE 'song_request|Bot .*connected|加载|Loaded|WS|error|Error' | tail -12
echo "===== 完 ====="
