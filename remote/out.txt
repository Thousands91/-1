--- 命令 ---
echo "==任务287729 重启bot加载关键词定版=="
OLD=""
for p in /proc/[0-9]*; do
  if [ "$(cat $p/comm 2>/dev/null)" = "python" ] && grep -qa "main.py" "$p/cmdline" 2>/dev/null; then
    OLD="$OLD ${p#/proc/}"
  fi
done
echo "旧bot进程:$OLD"
for pid in $OLD; do kill $pid 2>/dev/null; done
sleep 6
for pid in $OLD; do [ -d /proc/$pid ] && kill -9 $pid 2>/dev/null; done
sleep 3
cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py >> /root/bot.log 2>&1 &
sleep 18
FINAL=""
for p in /proc/[0-9]*; do
  if [ "$(cat $p/comm 2>/dev/null)" = "python" ] && grep -qa "main.py" "$p/cmdline" 2>/dev/null; then
    FINAL="$FINAL ${p#/proc/}"
  fi
done
echo "新bot进程:$FINAL (应为1个)"
grep -c "Succeeded to load" /root/bot.log
grep -E "Failed to load" /root/bot.log | head -3
tail -3 /root/bot.log
