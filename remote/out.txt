import requests
UA = {"User-Agent": "Mozilla/5.0 (Linux; Android 13) AppleWebKit/537.36 Chrome/120 Mobile Safari/537.36",
      "Accept-Language": "zh-CN,zh;q=0.9"}
r = requests.get("https://www.sogou.com/web", params={"query": "齐豫 是谁 歌手"}, headers=UA, timeout=10)
print("搜狗状态:", r.status_code, "长度:", len(r.text))
import re
blocked = any(w in r.text for w in ("安全验证","验证码","captcha"))
print("被风控:", blocked)
titles = re.findall(r"<h3[^>]*>(.*?)</h3>", r.text, re.S)[:3]
for t in titles:
    t2 = re.sub(r"<[^>]+>", "", t)
    print("结果:", t2[:50])
r2 = requests.get("https://www.baidu.com/s", params={"wd": "齐豫 歌手"}, headers=UA, timeout=10)
print("百度状态:", r2.status_code, "被风控:", any(w in r2.text for w in ("安全验证","验证码")))
