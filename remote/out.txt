echo -n "bash -l会话数: "; ps -eo cmd | grep -c '[b]ash -l'

ps -o pid,ppid,cmd -p 18783 2>/dev/null

echo "--- [5] wake-lock进程 ---"
pgrep -af 'wake-lock' || echo "无独立wake-lock进程(正常,它是申请锁后退出)"
