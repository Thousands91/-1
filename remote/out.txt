--- 命令 ---
echo "STEP1 下载"
cd /root/bot/bot
TOKEN=$(grep -o '[^:]*@github.com' /root/.git-credentials | head -1 | sed 's/@github.com//')
V=$(date +%s)$RANDOM
curl -s --max-time 60 -H "Authorization: token $TOKEN" -L "https://raw.githubusercontent.com/Thousands91/qq-bot/main/bot/plugins/ai_chat.py?v=$V" -o plugins/ai_chat.py
echo "STEP2 验证"
ls -la plugins/ai_chat.py
grep -c "_hist_text" plugins/ai_chat.py
python3 -c "import ast; ast.parse(open('plugins/ai_chat.py').read()); print('语法OK')"
echo "STEP3 重启bot"
kill $(pgrep -f "venv/bin/python main.py") 2>/dev/null; sleep 4
nohup /root/bot/venv/bin/python main.py >> bot.log 2>&1 &
sleep 15
pgrep -af "main.py" || echo "启动失败"
grep -a "connected" bot.log | tail -1
grep -a "21:5\|21:4" bot.log | grep -aiE "error|traceback" | tail -3 || echo "无报错"
