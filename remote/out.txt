cat > /root/azar_guard.sh << 'GUARDEOF'
#!/bin/bash
# 阿萨尔掉线守护 v1.1 (2026-11-02)
# v1.1 修复: 被踢后二维码变旧会被误判普通断线而白重启 —— 加"被踢标记"
# 规则1: QQ进程死了且无近期二维码 -> 自动拉起
# 规则2: WS断开+会话在(重启后未出新码) -> 自动重启NapCat免扫码恢复
# 规则3: 重启后仍出新款二维码(=真被踢) -> 记被踢标记,6小时内不再自动重启
# 规则4: 每小时最多重启3次,重启后冷却5分钟;登录成功(connected)自动清被踢标记
LOG=/root/bot.log
QR=/root/Napcat/opt/QQ/resources/app/app_launcher/napcat/cache/qrcode.png
STATE=/root/guard_status.txt
LAST_RESTART=/root/.guard_last
CNT_FILE=/root/.guard_cnt
KICKED=/root/.guard_kicked
TS() { stat -c %Y "$1" 2>/dev/null || echo 0; }
start_qq() {
  cd /root && nohup xvfb-run -a /root/Napcat/opt/QQ/qq --no-sandbox >> /root/napcat-restart.log 2>&1 &
}
can_restart() {
  NOW=$(date +%s)
  LR=$(cat "$LAST_RESTART" 2>/dev/null || echo 0)
  [ $(( NOW - LR )) -lt 300 ] && return 1
  H=$(date +%H)
  read C H2 < "$CNT_FILE" 2>/dev/null || { C=0; H2=$H; }
  [ "$H" != "$H2" ] && C=0
  [ "$C" -ge 3 ] && return 1
  echo "$((C+1)) $H" > "$CNT_FILE"
  echo "$NOW" > "$LAST_RESTART"
  return 0
}
echo "$(date '+%m-%d %H:%M:%S') 守护v1.1启动" >> "$STATE"
while true; do
  sleep 60
  LAST=$(grep "WebSocket for Bot 3427943187" "$LOG" 2>/dev/null | tail -1)
  case "$LAST" in
    *connected*)
      [ -f "$KICKED" ] && rm -f "$KICKED" && echo "$(date '+%m-%d %H:%M:%S') 已重新上线,清除被踢标记" >> "$STATE"
      continue
      ;;
  esac
  QQ_N=$(pgrep -fc "opt/QQ/qq --no-sandbox" 2>/dev/null)
  if [ "$QQ_N" = "0" ] || [ -z "$QQ_N" ]; then
    AGE=999999
    [ -f "$QR" ] && AGE=$(( $(date +%s) - $(TS "$QR") ))
    if [ $AGE -gt 600 ]; then
      if can_restart; then
        echo "$(date '+%m-%d %H:%M:%S') QQ进程消失,自动拉起" >> "$STATE"
        start_qq
      fi
    fi
    continue
  fi
  # QQ活着但WS不在connected态
  if [ -f "$KICKED" ]; then
    K=$(cat "$KICKED")
    NOW=$(date +%s)
    if [ $(( NOW - K )) -gt 21600 ]; then
      rm -f "$KICKED"
      echo "$(date '+%m-%d %H:%M:%S') 被踢标记超6小时,允许再试" >> "$STATE"
    else
      continue
    fi
  fi
  AGE=999999
  [ -f "$QR" ] && AGE=$(( $(date +%s) - $(TS "$QR") ))
  LR=$(cat "$LAST_RESTART" 2>/dev/null || echo 0)
  QRT=$(TS "$QR")
  if [ $AGE -lt 300 ]; then
    # 新鲜二维码: 刚被踢或刚重启出来,先标记(若此码在最近一次重启之后出现,基本坐实被踢)
    if [ "$QRT" -gt "$LR" ] && [ "$LR" != "0" ]; then
      echo "$(date +%s)" > "$KICKED"
      echo "$(date '+%m-%d %H:%M:%S') 重启后仍出新二维码: 确认被腾讯踢线,标记后等人工扫码" >> "$STATE"
    fi
    continue
  fi
  if can_restart; then
    echo "$(date '+%m-%d %H:%M:%S') WS断开,尝试重启NapCat免扫码恢复" >> "$STATE"
    pkill -f "opt/QQ/qq" 2>/dev/null
    sleep 3
    start_qq
  fi
done
GUARDEOF
chmod +x /root/azar_guard.sh
pkill -f azar_guard.sh 2>/dev/null
sleep 1
nohup bash /root/azar_guard.sh > /root/guard.log 2>&1 &
sleep 4
echo "--- v1.1 守护进程 ---"
pgrep -f azar_guard.sh | head -2
echo "--- 状态文件 ---"
tail -3 /root/guard_status.txt
