echo "=== ②部署登录态巡检(每5分钟记录,踢号时刻+在线周期从此有账) ==="
cat > /root/login_watch.sh << 'LW'
#!/bin/bash
# 登录态台账:每5分钟记一行logincheck结果,踢号时刻/在线时长/周期从此可回溯
LAST=""
while true; do
  S=$(bash /root/logincheck.sh 2>/dev/null | grep -oE "\[在线\]|\[掉线\]|\[未登录\]" | head -1)
  T=$(date '+%F %T')
  if [ "$S" != "$LAST" ]; then
    echo "$T 状态变化: ${LAST:-初始} -> ${S:-未知}" >> /root/login_state.log
    LAST="$S"
  else
    echo "$T $S" >> /root/login_state.log
  fi
  sleep 300
done
LW
chmod +x /root/login_watch.sh
setsid nohup /root/login_watch.sh > /dev/null 2>&1 < /dev/null &
sleep 3
pgrep -f login_watch.sh >/dev/null && echo "登录态巡检在跑" || echo "没起来"
tail -2 /root/login_state.log 2>/dev/null
echo "=== ③WS断连/闪断历史统计(bot.log) ==="
