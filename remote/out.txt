# 1. 智谱翻译（用 .env 真实key）
zp = os.getenv("OPENAI_API_KEY", "")
zpurl = os.getenv("OPENAI_BASE_URL", "")
zpmodel = os.getenv("OPENAI_MODEL", "")
print("1.翻译配置:", zpurl, zpmodel, "key前8:", zp[:8])
try:
    body = json.dumps({"model": zpmodel, "max_tokens":60,"temperature":0,
        "thinking":{"type":"disabled"},
        "messages":[{"role":"system","content":"你把中文搜索词翻译成二次元图库的英文检索词。只输出英文,最多3个不同说法的候选,用逗号分隔,全小写,用最朴素的常见英文单词或词组,不要自造复合词,不要解释。"},
                    {"role":"user","content":"奶奶"}]}).encode()
    req = urllib.request.Request(zpurl+"/chat/completions", data=body,
        headers={"Authorization":"Bearer "+zp,"Content-Type":"application/json","User-Agent":"t"})
    r = json.load(urllib.request.urlopen(req, timeout=20))
    print("   翻译结果:", r["choices"][0]["message"]["content"])
except Exception as e:
    print("   翻译失败:", type(e).__name__, str(e)[:200])

    try:
        url = f"https://tbib.org/autocomplete.php?q={urllib.parse.quote(q)}"

        vals = [(html.unescape(str(i.get('value',''))), html.unescape(str(i.get('label','')))) for i in (arr or [])[:5]]
        print(f"2.autocomplete q={q}: {vals}")

        print(f"2.autocomplete q={q} 失败:", type(e).__name__, str(e)[:150])
