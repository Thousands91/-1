--- 命令 ---
PREFIX=/data/data/com.termux/files/usr
U=https://mirrors.tuna.tsinghua.edu.cn/termux/apt/termux-main
F=$(python3 -c "
import re
t=open('/tmp/termux_packages.idx').read()
for blk in t.split('\n\n'):
    if re.search(r'^Package: libprotobuf$', blk, re.M):
        m=re.search(r'^Filename: (.+)$', blk, re.M)
        if m: print(m.group(1)); break
")
echo "libprotobuf路径: [$F]"
ENC=$(echo "$F" | sed 's/:/%3a/g')
curl -sL --max-time 120 -o /tmp/lp.deb "$U/$ENC" && ls -l /tmp/lp.deb && dpkg -x /tmp/lp.deb /tmp/lpx && echo 解包OK
dpkg -c /tmp/lp.deb | grep -E "\.so" | awk '{print $NF}' | head -8
find /tmp/lpx -name "*.so*" -exec cp -a {} /root/adblibs/ \; 2>/dev/null
echo "=== adb终试 ==="
export LD_LIBRARY_PATH=/root/adblibs:$PREFIX/lib
/root/adblibs/adb version 2>&1 | head -2
echo "=== pair命令 ==="
/root/adblibs/adb help 2>&1 | grep -E "pair|connect" | head -4
