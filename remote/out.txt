import json,urllib.request
req=urllib.request.Request("http://127.0.0.1:10800/",data=json.dumps({"type":"163","id":"284799"}).encode(),headers={"Content-Type":"application/json"})
c=json.loads(urllib.request.urlopen(req,timeout=20).read().decode(errors='replace'))
m=(c.get("meta") or {}).get("music") or {}
print("tag:",m.get("tag"))
print("jumpUrl:",m.get("jumpUrl"))
print("musicUrl:",m.get("musicUrl"))
print("token:",bool((c.get("config") or {}).get("token")))
PYEOF
