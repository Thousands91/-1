# 优化B替换+安全重启：备份→替换→后台延迟kill(PID文件)→删PID/锁→setsid重启→回传新PID
cp /root/remote_ctl.sh /root/remote_ctl.sh.bak_optb_1128 && echo "备份OK"
cp /tmp/remote_ctl_new.sh /root/remote_ctl.sh && chmod +x /root/remote_ctl.sh && echo "替换OK md5=$(md5sum /root/remote_ctl.sh | cut -d\  -f1)"
echo "落地语法=$(bash -n /root/remote_ctl.sh 2>&1 && echo OK || echo FAIL)"
OLD=$(cat /root/.remote_pid 2>/dev/null || echo 8689)
echo "准备重启:旧PID=$OLD"
# 后台延迟3秒执行重启链，让本任务先回传完成

echo "重启链已排入后台(延迟3s)"
echo "=== DEPLOY_B_RESTART_SENT ==="
