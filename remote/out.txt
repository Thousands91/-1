echo "===== [1] 取证日志现状(大小/行数/首末时间) ====="
ls -la /root/offline_forensics.log 2>/dev/null
wc -l /root/offline_forensics.log 2>/dev/null
head -1 /root/offline_forensics.log 2>/dev/null
tail -1 /root/offline_forensics.log 2>/dev/null
echo "===== [2] 有没有脚本/cron会清理或截断取证日志 ====="
grep -rn 'offline_forensics.log' /root/*.sh /root/*.py /etc/cron* /var/spool/cron 2>/dev/null | grep -iE 'rm |truncate|> */root/offline_forensics|: *>|cat /dev/null' | head -20
echo "(上面为空=没有清理脚本)"
echo "===== [3] crontab 里相关条目 ====="
crontab -l 2>/dev/null | grep -iE 'forensics|napcat|log' | head -10

ps -ef | grep '[o]ffline_watch.[s]h' | head -2
