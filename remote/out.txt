echo "===== 精确进程核查 $(date '+%F %T') ====="
echo "---逐个查/proc: comm=python且cmdline含main.py的才是真bot---"
for p in /proc/[0-9]*; do
  pid=${p##*/}
  comm=$(cat "$p/comm" 2>/dev/null)
  case "$comm" in
    python*|python3*)
      cmd=$(tr '\0' ' ' < "$p/cmdline" 2>/dev/null | cut -c1-90)
      echo "真python进程 PID=$pid comm=$comm cmd=$cmd"
      ;;
  esac
done
echo "---其中main.py进程数---"
CNT=0
for p in /proc/[0-9]*; do
  comm=$(cat "$p/comm" 2>/dev/null)
  case "$comm" in python*) grep -qa 'main.py' "$p/cmdline" 2>/dev/null && CNT=$((CNT+1));; esac
done
echo "main.py真进程数=$CNT"
echo "---遥控循环(方括号防自匹配)---"
pgrep -af 'remote_ctl.[s]h' | grep -v 'proot' | head -2 | cut -c1-70
echo "---登录态---"
tail -1 /root/login_state.log 2>/dev/null
echo "=====核查完====="
