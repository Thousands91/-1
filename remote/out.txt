echo "===== 坑159部署:重启bot(只重启bot不碰NapCat) ====="
echo "--- 当前bot PID ---"
ps -eo pid,args | grep '[m]ain.py' | awk '{print $1}'
if [ ! -f /root/.q159restart_armed ]; then
  touch /root/.q159restart_armed
  cat > /root/.q159restart.sh <<'SH'
#!/bin/bash
sleep 8
OLDPID=$(ps -eo pid,args | grep '[m]ain.py' | awk '{print $1}' | head -1)
echo "旧PID:$OLDPID"
[ -n "$OLDPID" ] && kill "$OLDPID"
sleep 3
cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py > /root/bot_restart_q159.log 2>&1 < /dev/null &
sleep 12
{
  echo "=== $(date '+%m-%d %H:%M:%S') 重启后 ==="
  echo "新PID: $(ps -eo pid,args | grep '[m]ain.py' | awk '{print $1}')"
  echo "--- 插件加载 ---"
  grep -E "Succeeded to load plugin|Failed to load plugin" /root/bot_restart_q159.log 2>/dev/null | tail -13
  echo "--- 失败数 ---"
  grep -c "Failed to load plugin" /root/bot_restart_q159.log 2>/dev/null
  echo "--- WS连接 ---"
  grep -iE "Bot .* connected" /root/bot_restart_q159.log 2>/dev/null | tail -2
  echo "--- 登录态 ---"
  tail -2 /root/login_state.log 2>/dev/null
  echo "--- 坑159新表述确认(设备实际文件) ---"
  grep -F -c '（引用一张图片，画面：{_qn}）' /root/bot/bot/plugins/ai_chat.py
} > /root/q159restart_verify.log 2>&1
rm -f /root/.q159restart_armed
SH
  chmod +x /root/.q159restart.sh
  setsid nohup bash /root/.q159restart.sh >/dev/null 2>&1 </dev/null &
  echo "延迟重启已挂(8s kill + 12s验证)"
fi
echo "=====完====="
