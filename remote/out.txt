import base64, json, urllib.request, shutil
files = {"ai_chat_face.py": "/root/bot/plugins/ai_chat.py",
         "image_search_multisource.py": "/root/bot/plugins/image_search.py"}
for f, dst in files.items():
    d = json.load(urllib.request.urlopen(f"https://api.github.com/repos/Thousands91/-1/contents/code/preview/{f}"))
    content = base64.b64decode(d["content"])
    bak = dst + ".bak_1102"
    shutil.copy(dst, bak)
    open(dst, "wb").write(content)
    print(f"OK {f} -> {dst} ({len(content)}字节, 旧版备份{bak})")
PYEOF
