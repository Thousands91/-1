[ -n "$OLD" ] && kill $OLD && sleep 2

echo "[6] 启动日志尾部(插件加载/Bot连接):"
tail -15 /root/bot/bot/bot.log 2>/dev/null | grep -E "song_request|Bot|connected|connected|加载|Loaded|ERROR|Traceback" | tail -8
echo "===== 完 ====="
