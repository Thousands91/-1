  adb shell device_config get activity_manager max_phantom_processes 2>&1
  echo "=== 待机桶(改前) ==="
  adb shell am get-standby-bucket com.termux 2>&1
  adb shell am set-standby-bucket com.termux active 2>&1
  echo "=== 待机桶(改后) ==="
  adb shell am get-standby-bucket com.termux 2>&1
  echo "=== doze状态与关闭 ==="
  adb shell dumpsys deviceidle disable 2>&1
  adb shell dumpsys deviceidle 2>/dev/null | grep mState | head -1
  echo "=== 电池/充电 ==="
  adb shell dumpsys battery 2>/dev/null | grep -E "powered|status|level" | head -4
else
