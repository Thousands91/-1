import urllib.request, time
urls = {
 "open-meteo forecast": "https://api.open-meteo.com/v1/forecast?latitude=39.9&longitude=116.4&current=temperature_2m&timezone=auto",
 "备用 wttr.in": "https://wttr.in/Beijing?format=j1",
 "备用 openweather(无key测连通)": "https://api.open-meteo.com/v1/forecast?latitude=23.1&longitude=113.3&current=temperature_2m",
}
for name,url in urls.items():
    ok=0; times=[]
    for i in range(3 if 'open-meteo' in name else 1):
        t=time.time()
        try:
            req=urllib.request.Request(url,headers={"User-Agent":"curl/8.0"})
            urllib.request.urlopen(req,timeout=12).read()
            ok+=1; times.append(f"{time.time()-t:.1f}s")
        except Exception as e:
            times.append(f"{type(e).__name__}:{time.time()-t:.1f}s")
    print(f"{name}: {ok}/3 成功 [{', '.join(times)}]")
PYEOF
echo "=====完====="
