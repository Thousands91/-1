python3 << 'PYEOF' 2>&1
p = '/root/bot/bot/plugins/ai_chat.py'
src = open(p, encoding='utf-8').read()
old = '语气克制、带点自嘲的清醒，不热血不中二，不卖惨。'
add = '\\n- 回复末尾的小符号（📖✨📚这些）别每条都带：十条里有两三条带就够，平时素着说，真有心境才点一个，不带也完全没问题。'
n = src.count(old)
print(f'锚点匹配: {n}处')
if n == 1:
    open(p, 'w', encoding='utf-8').write(src.replace(old, old + add))
    print('人设已更新')
PYEOF

/root/bot/venv/bin/python -m py_compile /root/bot/bot/plugins/ai_chat.py && echo "语法OK"

sleep 3

sleep 12
pgrep -f "venv/bin/python main.py" | head -1
tail -8 /root/bot/bot.log | grep -E "SUCCESS|ERROR|startup" | tail -3
