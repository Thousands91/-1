sleep 8

cd /root/bot/bot && setsid nohup /root/bot/venv/bin/python main.py > /root/bot_restart_162.log 2>&1 < /dev/null &
sleep 14
{
  echo "=== $(date '+%m-%d %H:%M:%S') 坑162重启后 ==="
  echo "新PID: $(ps -eo pid,args | grep '[m]ain.py' | awk '{print $1}')"
  echo "插件成功/失败: $(grep -c 'Succeeded to load plugin' /root/bot_restart_162.log 2>/dev/null) / $(grep -c 'Failed to load plugin' /root/bot_restart_162.log 2>/dev/null)"
  grep -iE 'Bot .* connected|connection open' /root/bot_restart_162.log 2>/dev/null | tail -3
  echo "登录态:"; tail -2 /root/login_state.log 2>/dev/null
  echo "坑162标记: 防棋步=$(grep -F -c '铁律：只有对方明确要说' /root/bot/bot/plugins/ai_chat.py) 立场=$(grep -F -c '立场一致铁律' /root/bot/bot/plugins/ai_chat.py)"
  echo "启动日志尾:"; tail -4 /root/bot_restart_162.log 2>/dev/null
} > /root/r162_verify.log 2>&1
rm -f /root/.r162_armed
SH
  chmod +x /root/.r162.sh
  setsid nohup bash /root/.r162.sh >/dev/null 2>&1 </dev/null &
  echo "延迟重启已挂(8s kill + 14s验证)"
fi
echo "=====完====="
