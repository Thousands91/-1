import asyncio, json
try:
    import websockets
except ImportError:
    print("no websockets lib"); raise SystemExit
async def main():
    uri="ws://127.0.0.1:8080/onebot/v11/ws"
    try:
        async with websockets.connect(uri, timeout=10) as ws:
            await ws.send(json.dumps({"action":"get_msg","params":{"message_id":1780830792},"echo":"g1"}))
            # 收几条,找echo=g1
            for _ in range(10):
                try:
                    raw=await asyncio.wait_for(ws.recv(),timeout=8)
                except asyncio.TimeoutError:
                    break
                d=json.loads(raw)
                if d.get("echo")=="g1":
                    data=d.get("data") or {}
                    msg=data.get("message") or []
                    print("消息段数:",len(msg) if isinstance(msg,list) else "非list")
                    if isinstance(msg,list):
                        for seg in msg:
                            if isinstance(seg,dict) and seg.get("type")=="image":
                                u=(seg.get("data") or {}).get("url","")
                                print("IMAGE_URL_FULL:",u)
                    break
    except Exception as e:
        print("WS失败:",type(e).__name__,str(e)[:100])
asyncio.run(main())
PYEOF
