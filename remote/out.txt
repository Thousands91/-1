echo "==任务287739 全脚本分类审计:揪出长驻但漏自启的=="
for f in ipwatch.sh azar_session_heal.sh watchdog2.sh watcher.sh x10_watchdog.sh remote_task.sh logincheck.sh asar_sync.sh p.sh ny.sh; do
  [ -f /root/$f ] || continue
  loop=$(grep -cE "while (true|:)|for \(\(" /root/$f)
  sleepn=$(grep -cE "^sleep |[0-9]+\s*(s|m|min)" /root/$f)
  live=$(pgrep -f "$f" | head -1)
  logf="/root/${f%.sh}.log"
  lg=$([ -f "$logf" ] && stat -c %y "$logf" | cut -d. -f1 || echo 无log)
  echo "$f | 循环=$loop | sleep=$sleepn | 进程=${live:-无} | log末次=$lg"
done
echo "--- ip.log末次写入确认 ---"
stat -c "%y" /root/ip.log 2>/dev/null; tail -2 /root/ip.log
