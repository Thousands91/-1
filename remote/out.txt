import json,urllib.request
def post(payload):
    req=urllib.request.Request("http://127.0.0.1:10800/",data=json.dumps(payload).encode(),headers={"Content-Type":"application/json"})
    return urllib.request.urlopen(req,timeout=20).read().decode()
# 场景A:只传id不传title(模拟NapCat官方卡契约)
a=post({"type":"163","id":"33737547"})
print("[A只传id]返回前300:",a[:300])
try:
    card=json.loads(a)
    m=card.get("meta",{}).get("music",{})
    print("  卡内 title=",repr(m.get("title"))," singer=",repr(m.get("desc"))," tag=",m.get("tag"))
except Exception as e: print("  解析失败",e)
# 场景B:带title
b=post({"type":"163","id":"33737547","title":"夜蝶","singer":"李艺彤"})
print("[B带title]返回前300:",b[:300])
try:
    card=json.loads(b); m=card.get("meta",{}).get("music",{})
    print("  卡内 title=",repr(m.get("title"))," singer=",repr(m.get("desc"))," tag=",m.get("tag"))
except Exception as e: print("  解析失败",e)
' 2>&1 | head -40
