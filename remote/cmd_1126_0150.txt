import sys, ast, asyncio, os
src=open('/root/bot/bot/plugins/ai_chat.py').read()
tree=ast.parse(src)
SHORT_SYS=None
for node in tree.body:
    if isinstance(node,ast.Assign):
        for t in node.targets:
            if isinstance(t,ast.Name) and t.id=='SHORT_FOLLOWUP_SYSTEM':
                SHORT_SYS=ast.literal_eval(node.value)
print("SHORT_SYS_LEN=",len(SHORT_SYS) if SHORT_SYS else None)
print("SHORT_SYS_HAS_e5=",'e5' in (SHORT_SYS or ''))
print("SHORT_SYS_HAS_齐豫=",'齐豫' in (SHORT_SYS or ''))
key=os.getenv('OPENAI_API_KEY','')
base=os.getenv('OPENAI_BASE_URL','https://api.siliconflow.cn/v1')
model=os.getenv('OPENAI_MODEL','Qwen/Qwen2.5-7B-Instruct')
print("KEY_OK=",bool(key)," BASE=",base," MODEL=",model)
from openai import AsyncOpenAI
msgs=[
  {"role":"user","content":"昨天那本书我翻到第三章了"},
  {"role":"assistant","content":"哦？讲到什么了，先生。"},
  {"role":"user","content":"？"},
]
full=[{"role":"system","content":SHORT_SYS}]+msgs
async def go():
    c=AsyncOpenAI(api_key=key,base_url=base,max_retries=0,timeout=15.0)
    try:
        r=await c.chat.completions.create(model=model,messages=full,max_tokens=120,temperature=0.7)
        print("ANSWER=",(r.choices[0].message.content or '')[:250])
    except Exception as e:
        print("CALL_ERR=",type(e).__name__,str(e)[:200])
    finally:
        await c.close()
asyncio.run(go())
